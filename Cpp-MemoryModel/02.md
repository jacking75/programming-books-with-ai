## 2.1 Relaxed Ordering

### 개념과 특징

`memory_order_relaxed`는 가장 약한 메모리 순서로, 원자성만 보장하고 다른 메모리 연산과의 순서는 전혀 보장하지 않는다. 이는 컴파일러와 CPU가 최대한의 최적화를 수행할 수 있게 한다.

**핵심 특징:**

```
Relaxed Ordering의 보장 사항:

✓ 원자성 (Atomicity)
  - 읽기/쓰기가 중간 상태 없이 완료됨
  - 데이터 레이스 방지

✗ 순서 보장 (Ordering)
  - 다른 메모리 연산과 순서 관계 없음
  - 스레드 간 동기화 없음

✗ 가시성 보장 (Visibility)
  - 언제 다른 스레드에 보일지 불확실
  - 각 스레드가 다른 순서로 관찰 가능
```

**메모리 연산의 재배치:**

```
원본 코드:                    가능한 실행 순서:

a.store(1, relaxed);          c.store(3, relaxed);  ← 재배치!
b.store(2, relaxed);          a.store(1, relaxed);
c.store(3, relaxed);          b.store(2, relaxed);

각 변수의 연산은 원자적이지만,
전체 순서는 보장되지 않는다.
```

### 사용 예제: 카운터 구현

**기본 카운터 - 단일 변수 증가:**

```cpp
#include <iostream>
#include <thread>
#include <atomic>
#include <vector>

// 예제 1: 단순 카운터 (적합한 사용 사례)
class SimpleCounter {
private:
    std::atomic<long long> count_{0};

public:
    void increment() {
        // relaxed는 카운터에 적합하다
        // 증가 자체의 원자성만 필요하고, 순서는 중요하지 않다
        count_.fetch_add(1, std::memory_order_relaxed);
    }

    long long get() const {
        // 읽기도 relaxed 사용
        return count_.load(std::memory_order_relaxed);
    }
};

void test_simple_counter() {
    SimpleCounter counter;
    std::vector<std::thread> threads;

    // 10개 스레드가 각각 100만 번 증가
    for (int i = 0; i < 10; ++i) {
        threads.emplace_back([&counter]() {
            for (int j = 0; j < 1'000'000; ++j) {
                counter.increment();
            }
        });
    }

    for (auto& t : threads) {
        t.join();
    }

    std::cout << "최종 카운트: " << counter.get() << std::endl;
    // 항상 10,000,000 출력 - 원자성 보장됨
}
```

```cpp
#include <iostream>

// 예제 2: 다중 통계 카운터
class StatisticsCollector {
private:
    std::atomic<uint64_t> total_requests_{0};
    std::atomic<uint64_t> successful_requests_{0};
    std::atomic<uint64_t> failed_requests_{0};
    std::atomic<uint64_t> total_bytes_{0};

public:
    void record_request(bool success, size_t bytes) {
        // 각 카운터는 독립적으로 증가
        // 상호 간의 순서는 중요하지 않다
        total_requests_.fetch_add(1, std::memory_order_relaxed);
        
        if (success) {
            successful_requests_.fetch_add(1, std::memory_order_relaxed);
        } else {
            failed_requests_.fetch_add(1, std::memory_order_relaxed);
        }
        
        total_bytes_.fetch_add(bytes, std::memory_order_relaxed);
    }

    void print_stats() const {
        // 읽기 시점에서의 근사값
        // 정확한 스냅샷은 아니지만 통계 목적으로는 충분
        uint64_t total = total_requests_.load(std::memory_order_relaxed);
        uint64_t success = successful_requests_.load(std::memory_order_relaxed);
        uint64_t failed = failed_requests_.load(std::memory_order_relaxed);
        uint64_t bytes = total_bytes_.load(std::memory_order_relaxed);

        std::cout << "=== 통계 ===" << std::endl;
        std::cout << "총 요청: " << total << std::endl;
        std::cout << "성공: " << success << std::endl;
        std::cout << "실패: " << failed << std::endl;
        std::cout << "총 바이트: " << bytes << std::endl;
        
        // 주의: success + failed != total 일 수 있음!
        // relaxed이므로 일관된 스냅샷이 아니다
    }
};

void test_statistics() {
    StatisticsCollector stats;
    std::vector<std::thread> threads;

    for (int i = 0; i < 5; ++i) {
        threads.emplace_back([&stats, i]() {
            for (int j = 0; j < 10000; ++j) {
                bool success = (j % 10) != 0;  // 90% 성공률
                size_t bytes = 100 + (j % 900);
                stats.record_request(success, bytes);
            }
        });
    }

    for (auto& t : threads) {
        t.join();
    }

    stats.print_stats();
}
```

```cpp
#include <thread>
#include <atomic>
#include <chrono>
#include <vector>

// 성능 벤치마크: relaxed vs seq_cst
void benchmark_memory_orders() {
    const int iterations = 10'000'000;
    const int num_threads = 4;

    // Relaxed
    {
        std::atomic<int> counter(0);
        auto start = std::chrono::high_resolution_clock::now();
        
        std::vector<std::thread> threads;
        for (int t = 0; t < num_threads; ++t) {
            threads.emplace_back([&counter, iterations]() {
                for (int i = 0; i < iterations; ++i) {
                    counter.fetch_add(1, std::memory_order_relaxed);
                }
            });
        }
        
        for (auto& t : threads) {
            t.join();
        }
        
        auto end = std::chrono::high_resolution_clock::now();
        auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);
        
        std::cout << "Relaxed: " << duration.count() << "ms" << std::endl;
        std::cout << "결과: " << counter.load() << std::endl;
    }

    // Sequential Consistency
    {
        std::atomic<int> counter(0);
        auto start = std::chrono::high_resolution_clock::now();
        
        std::vector<std::thread> threads;
        for (int t = 0; t < num_threads; ++t) {
            threads.emplace_back([&counter, iterations]() {
                for (int i = 0; i < iterations; ++i) {
                    counter.fetch_add(1, std::memory_order_seq_cst);
                }
            });
        }
        
        for (auto& t : threads) {
            t.join();
        }
        
        auto end = std::chrono::high_resolution_clock::now();
        auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);
        
        std::cout << "Seq_cst: " << duration.count() << "ms" << std::endl;
        std::cout << "결과: " << counter.load() << std::endl;
    }
}
```

### 주의사항과 함정

**함정 1: 동기화가 필요한 경우에 relaxed 사용**

```cpp
#include <iostream>
#include <thread>
#include <atomic>

// ❌ 잘못된 사용: 동기화 목적으로 relaxed 사용
std::atomic<bool> data_ready{false};
int data = 0;

void wrong_producer() {
    data = 42;
    data_ready.store(true, std::memory_order_relaxed);  // 문제!
}

void wrong_consumer() {
    while (!data_ready.load(std::memory_order_relaxed)) {
        // 대기
    }
    std::cout << "Data: " << data << std::endl;  // 0이 출력될 수 있음!
}

// ✓ 올바른 사용: acquire-release 사용
void correct_producer() {
    data = 42;
    data_ready.store(true, std::memory_order_release);  // 올바름
}

void correct_consumer() {
    while (!data_ready.load(std::memory_order_acquire)) {
        // 대기
    }
    std::cout << "Data: " << data << std::endl;  // 항상 42 출력
}
```

**동작 차이 시각화:**

```
Relaxed (잘못된 사용):

Thread 1 (Producer)          Thread 2 (Consumer)
    |                             |
    | data = 42                   |
    | (캐시에만 존재)             |
    |                             |
    | data_ready = true           |
    | (캐시에만 존재)             |
    |                             |
    |                             | while (!data_ready) {}
    |                             | // 메인 메모리에서 읽으면 false
    |                             | // 또는 true지만 data는 여전히 0
    |                             |
    | (나중에 메모리 동기화)      |

결과: 경쟁 조건 발생


Acquire-Release (올바른 사용):

Thread 1 (Producer)          Thread 2 (Consumer)
    |                             |
    | data = 42                   |
    |                             |
    | data_ready.store(release)   |
    | ──────────────────────────→ | (동기화 발생)
    |                             |
    |                             | data_ready.load(acquire)
    |                             | // data = 42가 보장됨
    |                             |
    |                             | cout << data  // 42
```

**함정 2: 일관된 스냅샷이 필요한 경우**

```cpp
#include <iostream>
#include <thread>
#include <atomic>

// ❌ 문제: 일관되지 않은 읽기
class BankAccount {
private:
    std::atomic<int> checking_{1000};
    std::atomic<int> savings_{2000};

public:
    void transfer_to_savings(int amount) {
        checking_.fetch_sub(amount, std::memory_order_relaxed);
        savings_.fetch_add(amount, std::memory_order_relaxed);

    int get_total_wrong() const {
        // 문제: 중간 상태를 볼 수 있음

        int s = savings_.load(std::memory_order_relaxed);
        return c + s;
        // checking에서 빠져나갔지만 savings에 아직 안 들어간 상태를 볼 수 있음!
    }
};

void test_inconsistent_snapshot() {
    BankAccount account;
    
    std::thread t1([&account]() {
        for (int i = 0; i < 1000; ++i) {
            account.transfer_to_savings(10);
        }
    });
    
    std::thread t2([&account]() {
        for (int i = 0; i < 100; ++i) {
            int total = account.get_total_wrong();
            if (total != 3000) {
                std::cout << "불일치 감지! 총액: " << total << std::endl;
                // 3000이 아닌 값이 출력될 수 있음
            }
        }
    });
    
    t1.join();
    t2.join();
}
```

**해결책:**

```cpp
// ✓ 해결책 1: 뮤텍스 사용
class BankAccountSafe1 {
private:
    int checking_{1000};
    int savings_{2000};
    mutable std::mutex mutex_;

public:
    void transfer_to_savings(int amount) {
        std::lock_guard<std::mutex> lock(mutex_);
        checking_ -= amount;
        savings_ += amount;
    }

    int get_total() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return checking_ + savings_;
    }
};

// ✓ 해결책 2: 단일 atomic 값 사용
class BankAccountSafe2 {
private:
    std::atomic<uint64_t> balances_{0};  // 상위 32비트: checking, 하위 32비트: savings

public:
    void transfer_to_savings(int amount) {
        uint64_t old_val, new_val;
        do {
            old_val = balances_.load(std::memory_order_relaxed);
            uint32_t checking = static_cast<uint32_t>(old_val >> 32);
            uint32_t savings = static_cast<uint32_t>(old_val & 0xFFFFFFFF);
            
            checking -= amount;
            savings += amount;
            
            new_val = (static_cast<uint64_t>(checking) << 32) | savings;
        } while (!balances_.compare_exchange_weak(old_val, new_val,
                                                   std::memory_order_relaxed));
    }

    int get_total() const {
        uint64_t val = balances_.load(std::memory_order_relaxed);
        uint32_t checking = static_cast<uint32_t>(val >> 32);
        uint32_t savings = static_cast<uint32_t>(val & 0xFFFFFFFF);
        return checking + savings;
    }
};
```

## 2.2 Acquire-Release Ordering

### Producer-Consumer 패턴

Acquire-Release 순서는 스레드 간 동기화의 핵심 메커니즘이다. Producer가 데이터를 준비하고 Consumer가 그것을 사용하는 패턴에서 가장 자주 사용된다.

**기본 개념:**

```
Release Store와 Acquire Load의 동기화:

Producer Thread              Consumer Thread
     |                            |
     | data1 = value1             |
     | data2 = value2             |
     | data3 = value3             |
     ↓                            |
[Release Store]                   |
  ready = true  ─────────────────→ [Acquire Load]
     |                            while (!ready)
     |                            ↓
     |                            | // 모든 data 값이
     |                            | // 보장됨
     |                            ↓
     |                         사용 가능
```

**단순 Producer-Consumer 구현:**

```cpp
#include <iostream>
#include <thread>
#include <atomic>
#include <string>

// 기본 Producer-Consumer
class SimpleMessage {
private:
    std::string data_;
    std::atomic<bool> ready_{false};

public:
    // Producer: 데이터 준비 및 발행
    void produce(const std::string& message) {
        data_ = message;  // (1) 데이터 준비
        
        // (2) release: 위의 모든 쓰기가 이 store 이전에 완료됨을 보장
        ready_.store(true, std::memory_order_release);
    }

    // Consumer: 데이터 소비
    std::string consume() {
        // (3) acquire: ready가 true면 모든 이전 쓰기를 볼 수 있음
        while (!ready_.load(std::memory_order_acquire)) {
            std::this_thread::yield();
        }
        
        // (4) data_는 완전히 초기화되어 있음이 보장됨
        return data_;
    }
};

void test_simple_message() {
    SimpleMessage msg;
    
    std::thread producer([&msg]() {
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
        msg.produce("Hello from producer!");
    });
    
    std::thread consumer([&msg]() {
        std::string data = msg.consume();
        std::cout << "Received: " << data << std::endl;
    });
    
    producer.join();
    consumer.join();
}
```

**복잡한 데이터 구조 전달:**

```cpp
#include <iostream>
#include <thread>
#include <atomic>
#include <vector>
#include <memory>

// 복잡한 데이터를 안전하게 전달
struct WorkItem {
    int id;
    std::string description;
    std::vector<int> data;
    
    WorkItem(int i, std::string desc, std::vector<int> d)
        : id(i), description(std::move(desc)), data(std::move(d)) {}
};

class WorkQueue {
private:
    WorkItem* item_{nullptr};  // 일반 포인터
    std::atomic<bool> ready_{false};

public:
    ~WorkQueue() {
        if (item_) {
            delete item_;
        }
    }

    // Producer: 작업 아이템 게시
    void publish(std::unique_ptr<WorkItem> item) {
        // (1) 새 아이템 준비
        WorkItem* new_item = item.release();
        
        // (2) 이전 아이템 정리 (있다면)
        WorkItem* old_item = item_;
        item_ = new_item;
        
        // (3) release: 모든 초기화가 완료됨
        ready_.store(true, std::memory_order_release);
        
        if (old_item) {
            delete old_item;
        }
    }

    // Consumer: 작업 아이템 가져오기
    std::unique_ptr<WorkItem> acquire() {
        // (4) acquire: ready가 true면 item_이 완전히 초기화됨
        while (!ready_.load(std::memory_order_acquire)) {
            std::this_thread::yield();
        }
        
        // (5) 안전하게 접근 가능
        ready_.store(false, std::memory_order_relaxed);
        return std::unique_ptr<WorkItem>(item_);
    }
};

void test_work_queue() {
    WorkQueue queue;
    
    std::thread producer([&queue]() {
        for (int i = 0; i < 5; ++i) {
            auto item = std::make_unique<WorkItem>(
                i,
                "Work item " + std::to_string(i),
                std::vector<int>{i, i*2, i*3}
            );
            
            std::this_thread::sleep_for(std::chrono::milliseconds(100));
            queue.publish(std::move(item));
            std::cout << "Published work item " << i << std::endl;
        }
    });
    
    std::thread consumer([&queue]() {
        for (int i = 0; i < 5; ++i) {
            auto item = queue.acquire();
            std::cout << "Processing: " << item->description << std::endl;
            std::cout << "  Data: ";
            for (int val : item->data) {
                std::cout << val << " ";
            }
            std::cout << std::endl;
        }
    });
    
    producer.join();
    consumer.join();
}
```

### 동기화 지점 이해하기

**Synchronizes-With 관계:**

```
동기화 지점 (Synchronization Point):

Thread A                         Thread B
   |                                |
   | X = 1                          |
   | Y = 2                          |
   | Z = 3                          |
   ↓                                |
[Release]                           |
flag.store(true) ────────────────→ [Acquire]
   |                              while (!flag.load())
   |                                ↓
   |                                | // X, Y, Z 모두
   |                                | // 보장됨
   |                                |

Synchronizes-With 관계:
- Release store가 Acquire load와 동기화됨
- Release 이전의 모든 메모리 쓰기가 Acquire 이후에 보임
```

**여러 변수의 동기화:**

```cpp
#include <iostream>
#include <thread>
#include <atomic>
#include <array>

// 여러 데이터의 동기화
class MultiDataPublisher {
private:
    std::array<int, 100> data_;
    std::atomic<int> published_count_{0};

public:
    // Producer: 데이터를 하나씩 준비
    void produce_data() {
        for (int i = 0; i < 100; ++i) {
            data_[i] = i * i;  // 데이터 준비
            
            // release: 이 인덱스까지의 모든 데이터가 준비됨
            published_count_.store(i + 1, std::memory_order_release);
            
            std::this_thread::sleep_for(std::chrono::microseconds(10));
        }
    }

    // Consumer: 준비된 데이터만큼 처리
    void consume_data() {
        int consumed = 0;
        
        while (consumed < 100) {
            // acquire: published_count 이하의 모든 데이터가 보임
            int available = published_count_.load(std::memory_order_acquire);
            
            // 새로 준비된 데이터 처리
            for (int i = consumed; i < available; ++i) {
                std::cout << "data[" << i << "] = " << data_[i] << std::endl;
            }
            
            consumed = available;
            std::this_thread::yield();
        }
    }
};

void test_multi_data() {
    MultiDataPublisher publisher;
    
    std::thread producer([&publisher]() {
        publisher.produce_data();
    });
    
    std::thread consumer([&publisher]() {
        publisher.consume_data();
    });
    
    producer.join();
    consumer.join();
}
```

**동기화 체인:**

```cpp
#include <iostream>
#include <thread>
#include <atomic>

// 동기화가 체인으로 전파됨
std::atomic<int> stage{0};
int data1 = 0, data2 = 0, data3 = 0;

void thread1() {
    data1 = 100;
    stage.store(1, std::memory_order_release);  // (A)
}

void thread2() {
    while (stage.load(std::memory_order_acquire) < 1) {}  // (B)
    // data1 = 100이 보장됨
    
    data2 = 200;
    stage.store(2, std::memory_order_release);  // (C)
}

void thread3() {
    while (stage.load(std::memory_order_acquire) < 2) {}  // (D)
    // data1 = 100, data2 = 200 모두 보장됨
    
    data3 = 300;
    stage.store(3, std::memory_order_release);  // (E)
}

void thread4() {
    while (stage.load(std::memory_order_acquire) < 3) {}  // (F)
    // data1, data2, data3 모두 보장됨
    
    std::cout << data1 << ", " << data2 << ", " << data3 << std::endl;
}

void test_sync_chain() {
    std::thread t1(thread1);
    std::thread t2(thread2);
    std::thread t3(thread3);
    std::thread t4(thread4);
    
    t1.join();
    t2.join();
    t3.join();
    t4.join();
}
```

**동기화 체인 시각화:**

```
동기화 전파 (Synchronization Propagation):

T1: data1=100 → [Release A] ─────────→ [Acquire B] T2
                                           ↓
T2:                                    data2=200
                                           ↓
                                      [Release C] ──→ [Acquire D] T3
                                                          ↓
T3:                                                   data3=300
                                                          ↓
                                                     [Release E] ──→ [Acquire F] T4
                                                                         ↓
T4:                                                                  모든 데이터 보임
                                                                     (100, 200, 300)

Happens-Before 관계:
data1 쓰기 → Release A → Acquire B → data2 쓰기 → Release C → Acquire D → data3 쓰기 → Release E → Acquire F → 모든 데이터 읽기
```

```cpp
#include <thread>
#include <atomic>

// 초기화 동기화
class LazyInitializer {
private:
    std::vector<int> data_;
    std::atomic<bool> initialized_{false};

public:
    void initialize() {
        // 무거운 초기화 작업
        data_.reserve(1000);
        for (int i = 0; i < 1000; ++i) {
            data_.push_back(i * i);
        }
        
        // release: 모든 초기화가 완료됨을 알림
        initialized_.store(true, std::memory_order_release);
    }

    const std::vector<int>& get_data() {
        // acquire: 초기화가 완료되었으면 data_에 안전하게 접근 가능
        while (!initialized_.load(std::memory_order_acquire)) {
            std::this_thread::yield();
        }
        
        return data_;
    }
};

void test_lazy_init() {
    LazyInitializer initializer;
    
    std::thread init_thread([&initializer]() {
        std::cout << "초기화 시작..." << std::endl;
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
        initializer.initialize();
        std::cout << "초기화 완료!" << std::endl;
    });
    
    std::thread user_thread([&initializer]() {
        const auto& data = initializer.get_data();
        std::cout << "데이터 사용: " << data.size() << " 항목" << std::endl;
        std::cout << "처음 10개: ";
        for (int i = 0; i < 10; ++i) {
            std::cout << data[i] << " ";
        }
        std::cout << std::endl;
    });
    
    init_thread.join();
    user_thread.join();
}
```

```cpp
#include <iostream>
#include <thread>
#include <atomic>
#include <string>

// 상태 기반 동기화
class TaskProcessor {
public:
    enum class State {
        Idle,
        Processing,
        Completed,
        Error
    };

private:
    std::string task_data_;
    std::string result_;
    std::atomic<State> state_{State::Idle};

public:
    void submit_task(const std::string& data) {
        task_data_ = data;
        // release: task_data_가 완전히 설정됨
        state_.store(State::Processing, std::memory_order_release);
    }

    void process() {
        // acquire: task_data_를 안전하게 읽을 수 있음
        while (state_.load(std::memory_order_acquire) != State::Processing) {
            std::this_thread::yield();
        }
        
        // 작업 처리
        result_ = "Processed: " + task_data_;
        
        // release: result_가 완전히 설정됨
        state_.store(State::Completed, std::memory_order_release);
    }

    std::string get_result() {
        // acquire: result_를 안전하게 읽을 수 있음
        while (state_.load(std::memory_order_acquire) != State::Completed) {
            std::this_thread::yield();
        }
        
        return result_;
    }
};

void test_state_machine() {
    TaskProcessor processor;
    
    std::thread submitter([&processor]() {
        processor.submit_task("Important Task");
        std::cout << "작업 제출됨" << std::endl;
    });
    
    std::thread worker([&processor]() {
        processor.process();
        std::cout << "작업 처리 완료" << std::endl;
    });
    
    std::thread retriever([&processor]() {
        std::string result = processor.get_result();
        std::cout << "결과: " << result << std::endl;
    });
    
    submitter.join();
    worker.join();
    retriever.join();
}
```

**패턴 3: 링 버퍼 (단일 Producer-Consumer)**

```cpp
#include <iostream>
#include <thread>
#include <atomic>
#include <array>
#include <optional>

// SPSC (Single Producer Single Consumer) 링 버퍼
template<typename T, size_t Size>
class SPSCQueue {
private:
    std::array<T, Size> buffer_;
    std::atomic<size_t> write_pos_{0};
    std::atomic<size_t> read_pos_{0};

public:
    // Producer: 항목 추가
    bool push(const T& item) {
        size_t current_write = write_pos_.load(std::memory_order_relaxed);
        size_t next_write = (current_write + 1) % Size;
        
        // 버퍼가 가득 찼는지 확인
        if (next_write == read_pos_.load(std::memory_order_acquire)) {
            return false;  // 가득 참
        }
        
        // 데이터 쓰기
        buffer_[current_write] = item;
        
        // release: buffer_에 데이터가 완전히 쓰임
        write_pos_.store(next_write, std::memory_order_release);
        return true;
    }

    // Consumer: 항목 제거
    std::optional<T> pop() {
        size_t current_read = read_pos_.load(std::memory_order_relaxed);
        
        // 버퍼가 비었는지 확인
        // acquire: write_pos_ 이전의 모든 쓰기를 볼 수 있음
        if (current_read == write_pos_.load(std::memory_order_acquire)) {
            return std::nullopt;  // 비어 있음
        }
        
        // 데이터 읽기 (완전히 초기화됨이 보장됨)
        T item = buffer_[current_read];
        
        // release: 읽기 완료
        read_pos_.store((current_read + 1) % Size, std::memory_order_release);
        return item;
    }
};

void test_spsc_queue() {
    SPSCQueue<int, 10> queue;
    std::atomic<bool> done{false};
    
    std::thread producer([&queue, &done]() {
        for (int i = 0; i < 100; ++i) {
            while (!queue.push(i)) {
                std::this_thread::yield();
            }
            std::cout << "Pushed: " << i << std::endl;
            std::this_thread::sleep_for(std::chrono::milliseconds(10));
        }
        done.store(true, std::memory_order_release);
    });
    
    std::thread consumer([&queue, &done]() {
        int count = 0;
        while (count < 100 || !done.load(std::memory_order_acquire)) {
            if (auto item = queue.pop()) {
                std::cout << "Popped: " << *item << std::endl;
                ++count;
            } else {
                std::this_thread::yield();
            }
        }
    });
    
    producer.join();
    consumer.join();
}
```

**패턴 4: 더블 버퍼링**

```cpp
#include <iostream>
#include <thread>
#include <atomic>
#include <vector>
#include <chrono>

// 더블 버퍼링을 사용한 데이터 교환
class DoubleBuffer {
private:
    std::vector<int> buffer0_;
    std::vector<int> buffer1_;
    std::atomic<int> current_write_buffer_{0};

public:
    // Writer: 현재 쓰기 버퍼에 데이터 추가
    void write(const std::vector<int>& data) {
        int write_idx = current_write_buffer_.load(std::memory_order_relaxed);
        
        if (write_idx == 0) {
            buffer0_ = data;
        } else {
            buffer1_ = data;
        }
        
        // release: 버퍼 데이터가 완전히 쓰임
        // 버퍼 전환
        current_write_buffer_.store(1 - write_idx, std::memory_order_release);
    }

    // Reader: 이전 쓰기 버퍼에서 데이터 읽기
    std::vector<int> read() {
        // acquire: 가장 최근에 완료된 버퍼의 데이터를 봄
        int write_idx = current_write_buffer_.load(std::memory_order_acquire);
        int read_idx = 1 - write_idx;  // 읽기는 쓰기 반대 버퍼
        
        return (read_idx == 0) ? buffer0_ : buffer1_;
    }
};

void test_double_buffer() {
    DoubleBuffer db;
    std::atomic<bool> done{false};
    
    std::thread writer([&db, &done]() {
        for (int frame = 0; frame < 10; ++frame) {
            std::vector<int> data;
            for (int i = 0; i < 5; ++i) {
                data.push_back(frame * 10 + i);
            }
            
            db.write(data);
            std::cout << "Frame " << frame << " written" << std::endl;
            std::this_thread::sleep_for(std::chrono::milliseconds(100));
        }
        done.store(true, std::memory_order_release);
    });
    
    std::thread reader([&db, &done]() {
        while (!done.load(std::memory_order_acquire)) {
            auto data = db.read();
            std::cout << "Read: ";
            for (int val : data) {
                std::cout << val << " ";
            }
            std::cout << std::endl;
            std::this_thread::sleep_for(std::chrono::milliseconds(150));
        }
    });
    
    writer.join();
    reader.join();
}
```

**메모리 순서 동작 비교:**

```
Acquire-Release 동작:

┌──────────────────────────────────────────────────────┐
│                  Producer Thread                      │
│  ┌──────────┐                                        │
│  │ data = X │  일반 쓰기                              │
│  └────┬─────┘                                        │
│       │                                               │
│       ↓                                               │
│  ┌─────────────────┐                                 │
│  │ ready = true    │  Release Store                  │
│  │ (memory_release)│                                 │
│  └────┬────────────┘                                 │
└───────┼──────────────────────────────────────────────┘
        │
        │ Synchronizes-With
        │
        ↓
┌───────┼──────────────────────────────────────────────┐
│  ┌────┴────────────┐                                 │
│  │ while (!ready)  │  Acquire Load                   │
│  │ (memory_acquire)│                                 │
│  └────┬────────────┘                                 │
│       │                                               │
│       ↓                                               │
│  ┌──────────┐                                        │
│  │ use data │  data = X가 보장됨                      │
│  └──────────┘                                        │
│                  Consumer Thread                      │
└──────────────────────────────────────────────────────┘

보장 사항:
✓ Producer의 모든 쓰기 → Release Store
✓ Release Store → Acquire Load (synchronizes-with)
✓ Acquire Load → Consumer의 모든 읽기
✓ 따라서: Producer의 쓰기 → Consumer의 읽기
```

## 2.3 Sequential Consistency

### 가장 강력한 보장

Sequential Consistency(순차 일관성)는 C++에서 제공하는 가장 강력한 메모리 순서 보장이다. 모든 스레드가 모든 원자 연산을 동일한 순서로 관찰하도록 보장한다.

**핵심 특징:**

```
Sequential Consistency의 보장:

1. 단일 전역 순서 (Single Total Order)
   - 모든 seq_cst 연산이 하나의 전역 순서를 가짐
   - 모든 스레드가 동일한 순서로 관찰

2. Program Order 유지
   - 각 스레드 내에서 프로그램 순서 유지
   
3. Acquire-Release 포함
   - Acquire와 Release의 모든 보장을 포함
```

**시각화:**

```
Sequential Consistency의 전역 순서:

Thread 1       Thread 2       Thread 3       전역적으로 보이는 순서
   |              |              |
   A              |              |           Global: A
   |              B              |           Global: A → B
   C              |              |           Global: A → B → C
   |              |              D           Global: A → B → C → D
   |              E              |           Global: A → B → C → D → E

모든 스레드가 A → B → C → D → E 순서로 관찰

반면 Relaxed는:
Thread 1이 보는 순서: A → C → B → D → E
Thread 2가 보는 순서: B → A → E → C → D
Thread 3이 보는 순서: D → A → B → C → E
```

**기본 예제:**

```cpp
#include <iostream>
#include <thread>
#include <atomic>

// Sequential Consistency 예제
std::atomic<int> x{0};
std::atomic<int> y{0};

void write_x_then_read_y() {
    x.store(1, std::memory_order_seq_cst);
    int r1 = y.load(std::memory_order_seq_cst);
    std::cout << "Thread 1: r1 = " << r1 << std::endl;
}

void write_y_then_read_x() {
    y.store(1, std::memory_order_seq_cst);
    int r2 = x.load(std::memory_order_seq_cst);
    std::cout << "Thread 2: r2 = " << r2 << std::endl;
}

void test_seq_cst() {
    // 이 테스트를 여러 번 실행해도
    // r1 == 0 && r2 == 0은 절대 발생하지 않음
    
    for (int i = 0; i < 1000; ++i) {
        x.store(0);
        y.store(0);
        
        std::thread t1(write_x_then_read_y);
        std::thread t2(write_y_then_read_x);
        
        t1.join();
        t2.join();
        
        // 가능한 결과:
        // r1=0, r2=1 (y.store가 x.store보다 먼저)
        // r1=1, r2=0 (x.store가 y.store보다 먼저)
        // r1=1, r2=1 (둘 다 관찰)
        // r1=0, r2=0은 불가능!
    }
}
```

**복잡한 동기화 예제:**

```cpp
#include <iostream>
#include <thread>
#include <atomic>
#include <vector>

// 다중 스레드 간의 복잡한 동기화
class SequentialBarrier {
private:
    std::atomic<int> counter_{0};
    std::atomic<bool> flags_[4] = {{false}, {false}, {false}, {false}};
    int data_[4] = {0, 0, 0, 0};

public:
    void worker(int id) {
        // 1단계: 데이터 준비
        data_[id] = (id + 1) * 100;
        
        // 2단계: 준비 완료 플래그 설정
        flags_[id].store(true, std::memory_order_seq_cst);
        
        // 3단계: 모든 스레드가 준비될 때까지 대기
        for (int i = 0; i < 4; ++i) {
            while (!flags_[i].load(std::memory_order_seq_cst)) {
                std::this_thread::yield();
            }
        }
        
        // 4단계: 모든 데이터 읽기 (모두 초기화됨이 보장됨)
        std::cout << "Thread " << id << " sees: ";
        for (int i = 0; i < 4; ++i) {
            std::cout << data_[i] << " ";
        }
        std::cout << std::endl;
    }
};

void test_barrier() {
    SequentialBarrier barrier;
    std::vector<std::thread> threads;
    
    for (int i = 0; i < 4; ++i) {
        threads.emplace_back([&barrier, i]() {
            barrier.worker(i);
        });
    }
    
    for (auto& t : threads) {
        t.join();
    }
    // 모든 스레드가 동일한 데이터를 봄: 100 200 300 400
}
```

```cpp
#include <iostream>

// 전역 순서가 중요한 경우
class EventLogger {
```

### 성능 트레이드오프

Sequential Consistency는 강력한 보장을 제공하지만 성능 비용이 있다.

**성능 벤치마크:**

```cpp
#include <iostream>
#include <thread>
#include <atomic>
#include <vector>
#include <chrono>

class MemoryOrderBenchmark {
private:
    static const int ITERATIONS = 10'000'000;
    static const int NUM_THREADS = 4;

public:
    // Relaxed 벤치마크
    static void benchmark_relaxed() {
        std::atomic<int> counter{0};
```

## 2.4 Acquire-Release vs Sequential Consistency

### 차이점 비교

**핵심 차이:**

```
Acquire-Release:
┌────────────────────────────────────────────┐
│ ✓ Pairwise 동기화                          │
│   - Release와 Acquire 쌍만 동기화          │
│ ✓ 더 약한 보장                             │
│   - 전역 순서 없음                         │
│ ✓ 더 빠른 성능                             │
│   - 최소한의 fence만 사용                  │
└────────────────────────────────────────────┘

Sequential Consistency:
┌────────────────────────────────────────────┐
│ ✓ 전역 순서                                │
│   - 모든 연산이 단일 순서                  │
│ ✓ 더 강한 보장                             │
│   - 모든 스레드가 동일 순서 관찰           │
│ ✓ 더 느린 성능                             │
│   - 더 많은 fence 필요                     │
└────────────────────────────────────────────┘
```

**실제 차이를 보여주는 예제:**

```cpp
#include <iostream>
#include <thread>
#include <atomic>

std::atomic<int> x{0}, y{0};
std::atomic<int> r1{0}, r2{0};

// Acquire-Release로는 방지할 수 없는 재배치
void test_acquire_release() {
    x.store(0); y.store(0);
    r1.store(0); r2.store(0);
    
    std::thread t1([]() {
        x.store(1, std::memory_order_release);
        r1.store(y.load(std::memory_order_acquire));
    });
    
    std::thread t2([]() {
        y.store(1, std::memory_order_release);
        r2.store(x.load(std::memory_order_acquire));
    });
    
    t1.join();
    t2.join();
    
    std::cout << "Acquire-Release: r1=" << r1 << ", r2=" << r2 << std::endl;
    // r1=0, r2=0이 가능! (전역 순서 없음)
}

// Sequential Consistency로 방지 가능
void test_seq_cst() {
    x.store(0); y.store(0);
    r1.store(0); r2.store(0);
    
    std::thread t1([]() {
        x.store(1, std::memory_order_seq_cst);
        r1.store(y.load(std::memory_order_seq_cst));
    });
    
    std::thread t2([]() {
        y.store(1, std::memory_order_seq_cst);
        r2.store(x.load(std::memory_order_seq_cst));
    });
    
    t1.join();
    t2.join();
    
    std::cout << "Seq_Cst: r1=" << r1 << ", r2=" << r2 << std::endl;
    // r1=0, r2=0은 불가능! (전역 순서 보장)
}

void run_comparison() {
    std::cout << "=== Acquire-Release vs Seq_Cst ===" << std::endl;
    
    for (int i = 0; i < 10; ++i) {
        test_acquire_release();
        test_seq_cst();
        std::cout << std::endl;
    }
}
```

**동작 차이 시각화:**

```
Acquire-Release의 가능한 실행:

Thread 1              Thread 2              관찰 가능한 순서들
   │                     │
   A: x=1                │                  T1: A → B
   │                     │                  T2: C → D
   B: r1=y               │
   │                     C: y=1             전역 순서 없음!
   │                     │
   │                     D: r2=x            가능한 순서:
                                           1) A → B → C → D  (r1=0, r2=1)
                                           2) C → D → A → B  (r1=1, r2=0)
                                           3) A → C → B → D  (r1=1, r2=1)
                                           4) A → C → D → B  (r1=0, r2=0) ← 가능!
                                           5) C → A → B → D  (r1=0, r2=1)
                                           6) C → A → D → B  (r1=0, r2=0) ← 가능!


Sequential Consistency의 실행:

Thread 1              Thread 2              단일 전역 순서
   │                     │
   A: x=1                │                  전역: A → B → C → D
   │                     │                  또는:  C → D → A → B
   B: r1=y               │                  또는:  A → C → B → D
   │                     C: y=1             등등...
   │                     │
   │                     D: r2=x            하지만 A → C → D → B는 불가능
                                           (B가 D 이후면 r1≠0이어야 함)

                                           따라서 r1=0, r2=0 불가능!
```

```cpp
        std::cout << "- Seq-Cst는 Relaxed 대비 30-100% 느림" << std::endl;
        std::cout << "- 플랫폼에 따라 차이 크게 변동" << std::endl;
    }
```

### 선택 가이드

**의사결정 트리:**

```cpp
#include <iostream>
#include <string>

class MemoryOrderSelector {
public:
    static std::string select_memory_order(
        bool needs_synchronization,
        bool multiple_variables,
        bool global_order_required,
        bool performance_critical
    ) {
        if (!needs_synchronization) {
            return "memory_order_relaxed";
        }
        
        if (global_order_required || multiple_variables) {
            if (performance_critical) {
                return "memory_order_seq_cst (프로파일 후 최적화 고려)";
            }
            return "memory_order_seq_cst";
        }
        
        if (performance_critical) {
            return "memory_order_acquire/release";
        }
        
        return "memory_order_seq_cst (안전)";
    }
    
    static void print_guide() {
        std::cout << R"(
메모리 순서 선택 가이드:

┌─────────────────────────────────────────────────────────┐
│ 1. 동기화가 필요한가?                                    │
│    NO  → memory_order_relaxed                           │
│    YES → 2번으로                                        │
└─────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────┐
│ 2. 여러 변수 간 순서가 중요한가?                         │
│    NO  → 3번으로                                        │
│    YES → memory_order_seq_cst                           │
└─────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────┐
│ 3. 단순 Producer-Consumer인가?                           │
│    YES → memory_order_acquire/release                   │
│    NO  → memory_order_seq_cst                           │
└─────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────┐
│ 4. 성능이 매우 중요한가?                                 │
│    YES → 프로파일링 후 최적화                            │
│    NO  → memory_order_seq_cst (안전)                    │
└─────────────────────────────────────────────────────────┘

    }
```

**실전 선택 예제:**

```cpp
#include <iostream>
#include <thread>
#include <atomic>

// 올바른 메모리 순서 선택 예제들

// 예제 1: 단순 통계 → Relaxed
class Statistics {
    std::atomic<uint64_t> count_{0};
public:
    void increment() {
        count_.fetch_add(1, std::memory_order_relaxed);  // ✓ 올바름
    }
};

// 예제 2: 플래그 동기화 → Acquire-Release
class FlaggedData {
    int data_;
    std::atomic<bool> ready_{false};
public:
    void produce(int value) {
        data_ = value;
        ready_.store(true, std::memory_order_release);  // ✓ 올바름
    }
    int consume() {
        while (!ready_.load(std::memory_order_acquire)) {}  // ✓ 올바름
        return data_;
    }
};

// 예제 3: 복잡한 상태 머신 → Seq-Cst
class StateMachine {
    std::atomic<int> state1_{0};
    std::atomic<int> state2_{0};
public:
    void complex_transition() {
        // 여러 상태가 얽혀있으면 seq_cst가 안전
        if (state1_.load(std::memory_order_seq_cst) == 1) {
            state2_.store(2, std::memory_order_seq_cst);  // ✓ 안전
        }
    }
};

int main() {
    MemoryOrderSelector::print_guide();
    ComprehensiveBenchmark::run_all();
    return 0;
}
```
