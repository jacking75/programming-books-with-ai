# 14장. 운영 환경 배포

## 1. 컨테이너화

### 1.1 Dockerfile 작성

```dockerfile
# 빌드 스테이지
FROM rust:1.70 as builder

WORKDIR /usr/src/app
COPY . .

# 릴리즈 빌드
RUN cargo build --release

# 실행 스테이지
FROM debian:bullseye-slim

# 필요한 런타임 의존성 설치
RUN apt-get update && apt-get install -y \
    ca-certificates \
    && rm -rf /var/lib/apt/lists/*

# 빌드된 바이너리 복사
COPY --from=builder /usr/src/app/target/release/my_tokio_app /usr/local/bin/

# 설정 파일 복사
COPY config/ /etc/my_tokio_app/

CMD ["my_tokio_app"]
```

### 1.2 컨테이너 구성 및 실행

```rust
use tokio::signal;

#[tokio::main]
async fn main() {
    // 컨테이너 환경 설정 로드
    let config = load_container_config().await?;

    // 그레이스풀 종료 처리
    tokio::spawn(async move {
        signal::ctrl_c().await.expect("Failed to listen for ctrl+c");
        println!("Shutting down gracefully...");
        shutdown_services().await;
    });

    // 애플리케이션 실행
    run_application(config).await
}

async fn load_container_config() -> Result<Config, Error> {
    let config_path = std::env::var("CONFIG_PATH")
        .unwrap_or_else(|_| "/etc/my_tokio_app/config.yaml".to_string());

    Config::from_file(&config_path).await
}
```

## 2. 클라우드 배포

### 2.1 AWS 배포 예제

```rust
use aws_sdk_ecs::{Client as EcsClient, Error};
use aws_config::Config;

struct CloudDeployer {
    ecs_client: EcsClient,
    config: DeploymentConfig,
}

impl CloudDeployer {
    async fn new() -> Result<Self, Error> {
        let config = aws_config::load_from_env().await;
        let ecs_client = EcsClient::new(&config);

        Ok(CloudDeployer {
            ecs_client,
            config: DeploymentConfig::default(),
        })
    }

    async fn deploy_service(&self) -> Result<(), Error> {
        // ECS 서비스 업데이트
        let request = self.ecs_client
            .update_service()
            .cluster(&self.config.cluster_name)
            .service(&self.config.service_name)
            .desired_count(self.config.desired_count)
            .force_new_deployment(true);

        request.send().await?;

        // 배포 상태 모니터링
        self.monitor_deployment().await?;

        Ok(())
    }

    async fn monitor_deployment(&self) -> Result<(), Error> {
        loop {
            let response = self.ecs_client
                .describe_services()
                .cluster(&self.config.cluster_name)
                .services(&[self.config.service_name.clone()])
                .send()
                .await?;

            if let Some(service) = response.services().first() {
                if is_deployment_complete(service) {
                    break;
                }
            }

            tokio::time::sleep(Duration::from_secs(10)).await;
        }
        Ok(())
    }
}
```

## 3. 모니터링 설정

### 3.1 Prometheus 메트릭 설정

```rust
use prometheus::{Registry, Counter, Histogram, IntGauge};

struct Metrics {
    registry: Registry,
    request_counter: Counter,
    response_time: Histogram,
    active_connections: IntGauge,
}

impl Metrics {
    fn new() -> Result<Self, prometheus::Error> {
        let registry = Registry::new();

        let request_counter = Counter::new(
            "http_requests_total",
            "Total number of HTTP requests"
        )?;

        let response_time = Histogram::with_opts(
            prometheus::HistogramOpts::new(
                "http_response_time_seconds",
                "HTTP response time in seconds"
            )
        )?;

        let active_connections = IntGauge::new(
            "active_connections",
            "Number of active connections"
        )?;

        registry.register(Box::new(request_counter.clone()))?;
        registry.register(Box::new(response_time.clone()))?;
        registry.register(Box::new(active_connections.clone()))?;

        Ok(Metrics {
            registry,
            request_counter,
            response_time,
            active_connections,
        })
    }

    async fn start_metrics_server(metrics: Arc<Metrics>) {
        let app = Router::new()
            .route("/metrics", get(|| async move {
                let encoder = TextEncoder::new();
                let metric_families = metrics.registry.gather();
                let mut buffer = vec![];
                encoder.encode(&metric_families, &mut buffer).unwrap();
                String::from_utf8(buffer).unwrap()
            }));

        axum::Server::bind(&"0.0.0.0:9091".parse().unwrap())
            .serve(app.into_make_service())
            .await
            .unwrap();
    }
}
```

## 4. 로그 관리

### 4.1 구조화된 로깅 구현

```rust
use tracing::{info, error, Level};
use tracing_subscriber::{FmtSubscriber, EnvFilter};
use serde_json::json;

struct LogManager {
    log_path: PathBuf,
    log_level: Level,
}

impl LogManager {
    async fn init(&self) -> Result<(), Error> {
        // JSON 포맷 로거 설정
        let file_appender = tracing_appender::rolling::RollingFileAppender::new(
            tracing_appender::rolling::RollingFileAppender::builder()
                .rotation(tracing_appender::rolling::Rotation::DAILY)
                .filename_prefix("app")
                .filename_suffix("log")
                .build(&self.log_path)?
        );

        let subscriber = FmtSubscriber::builder()
            .with_env_filter(EnvFilter::from_default_env()
                .add_directive(self.log_level.into()))
            .with_writer(file_appender)
            .with_thread_ids(true)
            .with_thread_names(true)
            .json()
            .flatten_event(true)
            .build();

        tracing::subscriber::set_global_default(subscriber)?;

        Ok(())
    }

    async fn log_event<T: serde::Serialize>(&self, event: &str, data: T) {
        info!(
            event = event,
            data = ?json!(data),
            timestamp = chrono::Utc::now().to_rfc3339()
        );
    }
}
```

## 5. CI/CD 파이프라인

### 5.1 GitHub Actions 워크플로우

```yaml
name: Rust CI/CD Pipeline

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

env:
  CARGO_TERM_COLOR: always

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
    - uses: actions/checkout@v2

    - name: Install Rust
      uses: actions-rs/toolchain@v1
      with:
        profile: minimal
        toolchain: stable

    - name: Run tests
      uses: actions-rs/cargo@v1
      with:
        command: test

    - name: Check formatting
      uses: actions-rs/cargo@v1
      with:
        command: fmt
        args: -- --check

  deploy:
    needs: test
    runs-on: ubuntu-latest
    if: github.ref == 'refs/heads/main'

    steps:
    - name: Configure AWS credentials
      uses: aws-actions/configure-aws-credentials@v1
      with:
        aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
        aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
        aws-region: us-east-1

    - name: Login to Amazon ECR
      id: login-ecr
      uses: aws-actions/amazon-ecr-login@v1

    - name: Build and push Docker image
      env:
        ECR_REGISTRY: ${{ steps.login-ecr.outputs.registry }}
        ECR_REPOSITORY: my-tokio-app
        IMAGE_TAG: ${{ github.sha }}
      run: |
        docker build -t $ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG .
        docker push $ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG

    - name: Deploy to ECS
      run: |
        aws ecs update-service --cluster production --service my-tokio-app --force-new-deployment
```

### 5.2 배포 자동화 스크립트

```rust
use tokio::process::Command;

struct Deployer {
    config: DeploymentConfig,
    environment: String,
}

impl Deployer {
    async fn deploy(&self) -> Result<(), Error> {
        // 사전 배포 검사
        self.pre_deployment_checks().await?;

        // 배포 실행
        self.run_deployment().await?;

        // 배포 후 검증
        self.post_deployment_verification().await?;

        Ok(())
    }

    async fn pre_deployment_checks(&self) -> Result<(), Error> {
        // 테스트 실행
        let status = Command::new("cargo")
            .args(&["test", "--release"])
            .status()
            .await?;

        if !status.success() {
            return Err(Error::TestsFailed);
        }

        // 보안 스캔
        self.run_security_scan().await?;

        Ok(())
    }

    async fn run_deployment(&self) -> Result<(), Error> {
        // Docker 이미지 빌드
        let tag = format!("{}:{}", self.config.image_name, self.config.version);

        Command::new("docker")
            .args(&["build", "-t", &tag, "."])
            .status()
            .await?;

        // 클라우드 배포
        let deployer = CloudDeployer::new().await?;
        deployer.deploy_service().await?;

        Ok(())
    }
}
```

핵심 포인트는 다음과 같다.

1. 컨테이너화
   - 다단계 빌드 최적화

   - 환경 설정 관리
   - 그레이스풀 종료

2. 클라우드 배포
   - 클라우드 플랫폼 통합
   - 인프라 자동화
   - 확장성 관리
   - 배포 모니터링

3. 모니터링 설정
   - 메트릭 수집
   - 알림 구성
   - 성능 모니터링
   - 상태 체크

4. 로그 관리
   - 구조화된 로깅
   - 로그 회전
   - 로그 집계
   - 로그 분석

5. CI/CD 파이프라인
   - 자동화된 테스트
   - 품질 검사
   - 자동 배포
   - 롤백 전략

구현 시 고려사항은 다음과 같다.

- 보안 설정
- 백업 전략
- 장애 복구
- 확장성
- 유지보수성
