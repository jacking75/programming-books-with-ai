# 13장. 확장성과 성능 최적화

## 1. 수평적 확장 (Horizontal Scaling)

### 1.1 클러스터 구성

```rust
use tokio::sync::mpsc;
use serde::{Serialize, Deserialize};

#[derive(Clone, Serialize, Deserialize)]
struct ClusterNode {
    id: String,
    addr: String,
    status: NodeStatus,
    last_heartbeat: DateTime<Utc>,
}

struct ClusterManager {
    nodes: Arc<RwLock<HashMap<String, ClusterNode>>>,
    tx: mpsc::Sender<ClusterEvent>,
}

impl ClusterManager {
    async fn new() -> Self {
        let (tx, mut rx) = mpsc::channel(100);
        let nodes = Arc::new(RwLock::new(HashMap::new()));
        let nodes_clone = nodes.clone();

        // 클러스터 이벤트 처리
        tokio::spawn(async move {
            while let Some(event) = rx.recv().await {
                let mut nodes = nodes_clone.write().await;
                match event {
                    ClusterEvent::NodeJoin(node) => {
                        nodes.insert(node.id.clone(), node);
                    }
                    ClusterEvent::NodeLeave(node_id) => {
                        nodes.remove(&node_id);
                    }
                    ClusterEvent::Heartbeat(node_id) => {
                        if let Some(node) = nodes.get_mut(&node_id) {
                            node.last_heartbeat = Utc::now();
                        }
                    }
                }
            }
        });

        ClusterManager { nodes, tx }
    }

    async fn join_cluster(&self, node: ClusterNode) -> Result<(), Error> {
        self.tx.send(ClusterEvent::NodeJoin(node)).await?;
        Ok(())
    }
}
```

### 1.2 작업 분산

```rust
struct WorkDistributor {
    cluster: Arc<ClusterManager>,
    work_queue: Arc<RwLock<VecDeque<Work>>>,
}

impl WorkDistributor {
    async fn distribute_work(&self) -> Result<(), Error> {
        let nodes = self.cluster.nodes.read().await;
        let mut work_queue = self.work_queue.write().await;

        while let Some(work) = work_queue.pop_front() {
            let available_nodes: Vec<_> = nodes.values()
                .filter(|node| node.status == NodeStatus::Ready)
                .collect();

            if available_nodes.is_empty() {
                work_queue.push_front(work);
                return Err(Error::NoAvailableNodes);
            }

            // 작업 할당 로직
            let selected_node = select_node(&available_nodes, &work)?;
            send_work_to_node(selected_node, work).await?;
        }

        Ok(())
    }
}
```

```rust
struct ResourceManager {
    cpu_cores: usize,
    memory_limit: usize,
    runtime_config: RuntimeConfig,
}

impl ResourceManager {
    async fn optimize_runtime(&mut self) -> Result<(), Error> {
        // CPU 코어 수에 따른 워커 스레드 조정
        let worker_threads = self.cpu_cores.max(1).min(32);

        let runtime = tokio::runtime::Builder::new_multi_thread()
            .worker_threads(worker_threads)
            .thread_name("optimized-worker")
            .thread_stack_size(3 * 1024 * 1024)  // 3MB 스택
            .enable_all()
            .build()?;

        // 메모리 제한 설정
        self.runtime_config.memory_limit = self.memory_limit;

        Ok(())
    }

    async fn monitor_resources(&self) -> ResourceMetrics {
        ResourceMetrics {
            cpu_usage: measure_cpu_usage().await?,
            memory_usage: measure_memory_usage().await?,
            thread_stats: collect_thread_statistics().await?,
        }
    }
}
```

## 3. 캐싱 전략

### 3.1 분산 캐시 구현

```rust
use tokio::time::{Duration, Instant};

struct CacheEntry<T> {
    value: T,
    expires_at: Instant,
}

struct DistributedCache<T> {
    local_cache: Arc<RwLock<HashMap<String, CacheEntry<T>>>>,
    remote_cache: Arc<dyn RemoteCache<T>>,
    ttl: Duration,
}

impl<T: Clone + Send + Sync + 'static> DistributedCache<T> {
    async fn get(&self, key: &str) -> Option<T> {
        // 로컬 캐시 확인
        if let Some(entry) = self.local_cache.read().await.get(key) {
            if entry.expires_at > Instant::now() {
                return Some(entry.value.clone());
            }
        }

        // 원격 캐시 확인
        if let Some(value) = self.remote_cache.get(key).await {
            self.set_local(key.to_string(), value.clone()).await;
            return Some(value);
        }

        None
    }

    async fn set_local(&self, key: String, value: T) {
        let entry = CacheEntry {
            value,
            expires_at: Instant::now() + self.ttl,
        };
        self.local_cache.write().await.insert(key, entry);
    }
}
```

### 3.2 캐시 일관성 관리

```rust
struct CacheConsistencyManager {
    cache: Arc<DistributedCache<Vec<u8>>>,
    invalidation_channel: mpsc::Receiver<InvalidationEvent>,
}

impl CacheConsistencyManager {
    async fn run(&mut self) {
        while let Some(event) = self.invalidation_channel.recv().await {
            match event {
                InvalidationEvent::KeyModified { key } => {
                    self.cache.invalidate(&key).await?;
                    self.notify_other_nodes(&key).await?;
                }
                InvalidationEvent::FullSync => {
                    self.perform_full_sync().await?;
                }
            }
        }
    }

    async fn perform_full_sync(&self) -> Result<(), Error> {
        let all_keys = self.cache.get_all_keys().await?;
        let mut outdated_keys = Vec::new();

        for key in all_keys {
            if !self.verify_consistency(&key).await? {
                outdated_keys.push(key);
            }
        }

        self.sync_keys(&outdated_keys).await?;
        Ok(())
    }
}
```

### 4.1 비동기 성능 최적화

```rust
struct PerformanceOptimizer {
    config: OptimizerConfig,
    metrics: Arc<Metrics>,
}

impl PerformanceOptimizer {
    async fn optimize_async_operations(&self) -> Result<(), Error> {
        // 백프레셔 설정
        self.configure_backpressure().await?;

        // 버퍼 크기 최적화
        self.optimize_buffer_sizes().await?;

        // 비동기 작업 배치 처리
        self.configure_batch_processing().await?;

        Ok(())
    }

    async fn configure_backpressure(&self) -> Result<(), Error> {
        let (tx, rx) = mpsc::channel(self.config.channel_buffer_size);

        // 배압 모니터링
        tokio::spawn(async move {
            while let Some(item) = rx.recv().await {
                if rx.capacity() < self.config.min_buffer_capacity {
                    self.metrics.record_backpressure().await;
                    tokio::time::sleep(Duration::from_millis(10)).await;
                }
                process_item(item).await?;
            }
        });

        Ok(())
    }
}
```

### 5.1 부하 테스트 프레임워크

```rust
struct LoadTester {
    config: LoadTestConfig,
    metrics_collector: Arc<MetricsCollector>,
}

impl LoadTester {
    async fn run_load_test(&self) -> TestResults {
        let start_time = Instant::now();
        let mut handles = Vec::new();

        // 동시 사용자 시뮬레이션
        for i in 0..self.config.concurrent_users {
            let metrics = self.metrics_collector.clone();
            let config = self.config.clone();

            handles.push(tokio::spawn(async move {
                simulate_user_behavior(i, config, metrics).await
            }));
        }

        // 결과 수집
        let mut results = Vec::new();
        for handle in handles {
            results.push(handle.await?);
        }

        TestResults {
            duration: start_time.elapsed(),
            requests_per_second: calculate_rps(&results),
            latency_stats: calculate_latency_stats(&results),
            error_rate: calculate_error_rate(&results),
        }
    }

    async fn simulate_user_behavior(
        user_id: usize,
        config: LoadTestConfig,
        metrics: Arc<MetricsCollector>,
    ) -> UserSimulationResult {
        let mut results = Vec::new();

        for _ in 0..config.requests_per_user {
            let start = Instant::now();
            let response = make_request(&config.target_url).await;
            let latency = start.elapsed();

            metrics.record_request(latency, response.status()).await;
            results.push(RequestResult { latency, response });

            // 사용자 행동 시뮬레이션을 위한 랜덤 지연
            tokio::time::sleep(Duration::from_millis(
                rand::thread_rng().gen_range(100..1000)
            )).await;
        }

        UserSimulationResult { user_id, results }
    }
}
```

### 5.2 메트릭 수집 및 분석

```rust
struct MetricsCollector {
    latencies: Arc<RwLock<Vec<Duration>>>,
    errors: Arc<AtomicUsize>,
    requests: Arc<AtomicUsize>,
}

impl MetricsCollector {
    async fn record_request(&self, latency: Duration, status: StatusCode) {
        self.latencies.write().await.push(latency);
        self.requests.fetch_add(1, Ordering::SeqCst);

        if !status.is_success() {
            self.errors.fetch_add(1, Ordering::SeqCst);
        }
    }

    async fn generate_report(&self) -> MetricsReport {
        let latencies = self.latencies.read().await;
        let total_requests = self.requests.load(Ordering::SeqCst);
        let total_errors = self.errors.load(Ordering::SeqCst);

        MetricsReport {
            total_requests,
            total_errors,
            error_rate: total_errors as f64 / total_requests as f64,
            avg_latency: calculate_average(&latencies),
            percentiles: calculate_percentiles(&latencies),
            throughput: calculate_throughput(total_requests, latencies.len()),
        }
    }
}
```
