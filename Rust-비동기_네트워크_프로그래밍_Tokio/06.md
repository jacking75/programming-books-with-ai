# 6장. 고성능 네트워킹

## 1. 버퍼 관리

### 1.1 버퍼의 기본 개념
버퍼는 데이터를 임시로 저장하는 메모리 공간으로, 네트워크 프로그래밍에서 매우 중요한 역할을 한다.

```rust
use bytes::{BytesMut, BufMut};

struct NetworkBuffer {
    read_buffer: BytesMut,
    write_buffer: BytesMut,
    buffer_size: usize,
}

impl NetworkBuffer {
    fn new(size: usize) -> Self {
        NetworkBuffer {
            read_buffer: BytesMut::with_capacity(size),
            write_buffer: BytesMut::with_capacity(size),
            buffer_size: size,
        }
    }

    fn write_data(&mut self, data: &[u8]) -> Result<(), Box<dyn std::error::Error>> {
        if self.write_buffer.len() + data.len() <= self.buffer_size {
            self.write_buffer.put_slice(data);
            Ok(())
        } else {
            Err("Buffer overflow".into())
        }
    }

    fn read_data(&mut self, size: usize) -> Option<BytesMut> {
        if size <= self.read_buffer.len() {
            Some(self.read_buffer.split_to(size))
        } else {
            None
        }
    }
}
```

### 1.2 버퍼 풀링
메모리 할당/해제 비용을 줄이기 위한 버퍼 풀 구현이다.

```rust
use std::sync::{Arc, Mutex};

struct BufferPool {
    pool: Arc<Mutex<Vec<BytesMut>>>,
    buffer_size: usize,
    max_buffers: usize,
}

impl BufferPool {
    fn new(buffer_size: usize, max_buffers: usize) -> Self {
        BufferPool {
            pool: Arc::new(Mutex::new(Vec::new())),
            buffer_size,
            max_buffers,
        }
    }

    fn acquire(&self) -> Option<BytesMut> {
        let mut pool = self.pool.lock().unwrap();
        if let Some(buffer) = pool.pop() {
            Some(buffer)
        } else if pool.len() < self.max_buffers {
            Some(BytesMut::with_capacity(self.buffer_size))
        } else {
            None
        }
    }

    fn release(&self, mut buffer: BytesMut) {
        buffer.clear();
        let mut pool = self.pool.lock().unwrap();
        if pool.len() < self.max_buffers {
            pool.push(buffer);
        }
    }
}
```
  

## 2. 메모리 최적화

### 2.1 메모리 할당 최적화

```rust
use std::mem::MaybeUninit;

struct OptimizedBuffer<const N: usize> {
    data: [MaybeUninit<u8>; N],
    len: usize,
}

impl<const N: usize> OptimizedBuffer<N> {
    fn new() -> Self {
        OptimizedBuffer {
            data: unsafe { MaybeUninit::uninit().assume_init() },
            len: 0,
        }
    }

    fn write(&mut self, data: &[u8]) -> Result<(), &'static str> {
        if self.len + data.len() > N {
            return Err("Buffer full");
        }

        for (i, &byte) in data.iter().enumerate() {
            self.data[self.len + i].write(byte);
        }
        self.len += data.len();
        Ok(())
    }
}
```

### 2.2 메모리 정렬과 캐시 최적화

```rust
#[repr(align(64))]  // 캐시 라인 크기에 맞춤
struct CacheAlignedBuffer {
    data: Vec<u8>,
    position: usize,
}

impl CacheAlignedBuffer {
    fn new(size: usize) -> Self {
        CacheAlignedBuffer {
            data: Vec::with_capacity(size),
            position: 0,
        }
    }

    fn write_aligned(&mut self, data: &[u8]) {
        // 64바이트 경계에 맞춰 데이터 쓰기
        let alignment = 64;
        let padding = (alignment - (self.position % alignment)) % alignment;
        self.position += padding;
        self.data.extend_from_slice(data);
        self.position += data.len();
    }
}
```
  

## 3. Zero-copy 구현

### 3.1 기본 Zero-copy 전송

```rust
use std::fs::File;
use std::os::unix::io::AsRawFd;
use nix::sys::sendfile::sendfile;

struct ZeroCopyTransfer {
    file: File,
}

impl ZeroCopyTransfer {
    fn new(file: File) -> Self {
        ZeroCopyTransfer { file }
    }

    fn transfer_to_socket(&self, socket: &std::net::TcpStream, count: usize) -> Result<usize, std::io::Error> {
        let fd_in = self.file.as_raw_fd();
        let fd_out = socket.as_raw_fd();

        match sendfile(fd_out, fd_in, None, count) {
            Ok(written) => Ok(written),
            Err(e) => Err(std::io::Error::new(std::io::ErrorKind::Other, e)),
        }
    }
}
```

### 3.2 메모리 맵핑을 이용한 Zero-copy

```rust
use memmap::MmapOptions;

struct MemoryMappedFile {
    mmap: memmap::Mmap,
}

impl MemoryMappedFile {
    fn new(file: &File) -> std::io::Result<Self> {
        let mmap = unsafe { MmapOptions::new().map(file)? };
        Ok(MemoryMappedFile { mmap })
    }

    fn read_at(&self, offset: usize, len: usize) -> Option<&[u8]> {
        if offset + len <= self.mmap.len() {
            Some(&self.mmap[offset..offset + len])
        } else {
            None
        }
    }
}
```

## 4. 네트워크 성능 측정

### 4.1 성능 측정 도구 구현

```rust
use std::time::{Instant, Duration};

struct PerformanceMetrics {
    start_time: Instant,
    bytes_transferred: usize,
    operation_count: u64,
}

impl PerformanceMetrics {
    fn new() -> Self {
        PerformanceMetrics {
            start_time: Instant::now(),
            bytes_transferred: 0,
            operation_count: 0,
        }
    }

    fn record_operation(&mut self, bytes: usize) {
        self.bytes_transferred += bytes;
        self.operation_count += 1;
    }

    fn get_metrics(&self) -> (f64, f64) {
        let duration = self.start_time.elapsed().as_secs_f64();
        let throughput = self.bytes_transferred as f64 / duration / 1_000_000.0; // MB/s
        let ops_per_second = self.operation_count as f64 / duration;

        (throughput, ops_per_second)
    }
}
```

### 4.2 레이턴시 측정

```rust
struct LatencyMeasurement {
    samples: Vec<Duration>,
    current_operation: Option<Instant>,
}

impl LatencyMeasurement {
    fn new() -> Self {
        LatencyMeasurement {
            samples: Vec::new(),
            current_operation: None,
        }
    }

    fn start_operation(&mut self) {
        self.current_operation = Some(Instant::now());
    }

    fn end_operation(&mut self) {
        if let Some(start) = self.current_operation.take() {
            let duration = start.elapsed();
            self.samples.push(duration);
        }
    }

    fn get_statistics(&self) -> (Duration, Duration, Duration) {
        let mut sorted_samples = self.samples.clone();
        sorted_samples.sort();

        let avg = sorted_samples.iter().sum::<Duration>() / sorted_samples.len() as u32;
        let median = sorted_samples[sorted_samples.len() / 2];
        let p99 = sorted_samples[(sorted_samples.len() as f64 * 0.99) as usize];

        (avg, median, p99)
    }
}
```
  

## 5. 병목 현상 분석과 해결

### 5.1 병목 현상 분석 도구

```rust
struct BottleneckAnalyzer {
    cpu_usage: Vec<f64>,
    memory_usage: Vec<usize>,
    io_wait: Vec<Duration>,
    network_usage: Vec<usize>,
}

impl BottleneckAnalyzer {
    fn new() -> Self {
        BottleneckAnalyzer {
            cpu_usage: Vec::new(),
            memory_usage: Vec::new(),
            io_wait: Vec::new(),
            network_usage: Vec::new(),
        }
    }

    fn record_metrics(&mut self, cpu: f64, memory: usize, io: Duration, network: usize) {
        self.cpu_usage.push(cpu);
        self.memory_usage.push(memory);
        self.io_wait.push(io);
        self.network_usage.push(network);
    }

    fn analyze(&self) -> Vec<&'static str> {
        let mut bottlenecks = Vec::new();

        // CPU 병목 확인
        if self.cpu_usage.iter().any(|&usage| usage > 80.0) {
            bottlenecks.push("High CPU usage detected");
        }

        // 메모리 병목 확인
        if self.memory_usage.iter().any(|&usage| usage > 90 * 1024 * 1024) {
            bottlenecks.push("High memory usage detected");
        }

        // I/O 대기 시간 분석
        if self.io_wait.iter().any(|wait| wait > &Duration::from_millis(100)) {
            bottlenecks.push("High I/O wait times detected");
        }

        bottlenecks
    }
}
```

### 5.2 성능 최적화 예제

```rust
struct NetworkOptimizer {
    buffer_pool: BufferPool,
    metrics: PerformanceMetrics,
    latency_measurement: LatencyMeasurement,
}

impl NetworkOptimizer {
    fn new(buffer_size: usize, max_buffers: usize) -> Self {
        NetworkOptimizer {
            buffer_pool: BufferPool::new(buffer_size, max_buffers),
            metrics: PerformanceMetrics::new(),
            latency_measurement: LatencyMeasurement::new(),
        }
    }

    async fn optimized_send(&mut self, data: &[u8], socket: &mut tokio::net::TcpStream) -> std::io::Result<()> {
        self.latency_measurement.start_operation();

        if let Some(mut buffer) = self.buffer_pool.acquire() {
            buffer.extend_from_slice(data);
            socket.write_all(&buffer).await?;
            self.buffer_pool.release(buffer);

            self.metrics.record_operation(data.len());
            self.latency_measurement.end_operation();
            Ok(())
        } else {
            Err(std::io::Error::new(
                std::io::ErrorKind::Other,
                "No available buffers"
            ))
        }
    }

    fn get_performance_report(&self) -> String {
        let (throughput, ops) = self.metrics.get_metrics();
        let (avg, median, p99) = self.latency_measurement.get_statistics();

        format!(
            "Performance Report:\n\
             Throughput: {:.2} MB/s\n\
             Operations/sec: {:.2}\n\
             Average Latency: {:?}\n\
             Median Latency: {:?}\n\
             P99 Latency: {:?}",
            throughput, ops, avg, median, p99
        )
    }
}
```

2. Zero-copy 기술을 통한 데이터 전송 효율화
3. 성능 측정 및 모니터링 방법

- 버퍼 크기는 애플리케이션의 특성에 맞게 조정
- 메모리 할당은 최소화하고 재사용 극대화
- 성능 측정은 실제 운영 환경과 유사한 조건에서 수행
- 병목 현상은 종합적인 관점에서 분석

위의 개념들을 적절히 적용하면 고성능의 네트워크 애플리케이션을 구현할 수 있다.
