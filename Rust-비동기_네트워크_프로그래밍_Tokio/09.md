## 1. TLS 구현

### 1.1 기본 TLS 서버/클라이언트 구현

```rust
use tokio::net::TcpStream;
use tokio_rustls::rustls::{self, ClientConfig, ServerConfig};
use tokio_rustls::{TlsConnector, TlsAcceptor};
use std::sync::Arc;

// TLS 클라이언트 구현
struct TlsClient {
    connector: TlsConnector,
    domain: String,
}

impl TlsClient {
    fn new(domain: String, root_cert_store: rustls::RootCertStore) -> Result<Self, Box<dyn Error>> {
        let config = ClientConfig::builder()
            .with_safe_defaults()
            .with_root_certificates(root_cert_store)
            .with_no_client_auth();

        Ok(TlsClient {
            connector: TlsConnector::from(Arc::new(config)),
            domain,
        })
    }

    async fn connect(&self, addr: String) -> Result<tokio_rustls::client::TlsStream<TcpStream>, Box<dyn Error>> {
        let stream = TcpStream::connect(addr).await?;
        let tls_stream = self.connector.connect(self.domain.as_str().try_into()?, stream).await?;
        Ok(tls_stream)
    }
}

// TLS 서버 구현
struct TlsServer {
    acceptor: TlsAcceptor,
}

impl TlsServer {
    fn new(cert_file: &[u8], key_file: &[u8]) -> Result<Self, Box<dyn Error>> {
        let cert_chain = vec![rustls::Certificate(cert_file.to_vec())];
        let key = rustls::PrivateKey(key_file.to_vec());

        let config = ServerConfig::builder()
            .with_safe_defaults()
            .with_no_client_auth()
            .with_single_cert(cert_chain, key)?;

        Ok(TlsServer {
            acceptor: TlsAcceptor::from(Arc::new(config))
        })
    }

    async fn accept(&self, stream: TcpStream) -> Result<tokio_rustls::server::TlsStream<TcpStream>, Box<dyn Error>> {
        let tls_stream = self.acceptor.accept(stream).await?;
        Ok(tls_stream)
    }
}
```

### 1.2 TLS 설정 관리

```rust
use std::fs::File;
use std::io::BufReader;

struct TlsConfig {
    cert_path: String,
    key_path: String,
    root_store: rustls::RootCertStore,
}

impl TlsConfig {
    fn new(cert_path: String, key_path: String) -> Result<Self, Box<dyn Error>> {
        let mut root_store = rustls::RootCertStore::empty();
        root_store.add_server_trust_anchors(
            webpki_roots::TLS_SERVER_ROOTS
                .0
                .iter()
                .map(|ta| {
                    rustls::OwnedTrustAnchor::from_subject_spki_name_constraints(
                        ta.subject,
                        ta.spki,
                        ta.name_constraints,
                    )
                })
        );

        Ok(TlsConfig {
            cert_path,
            key_path,
            root_store,
        })
    }

    fn load_certs(&self) -> Result<Vec<rustls::Certificate>, Box<dyn Error>> {
        let cert_file = File::open(&self.cert_path)?;
        let mut reader = BufReader::new(cert_file);
        let certs = rustls_pemfile::certs(&mut reader)?
            .iter()
            .map(|v| rustls::Certificate(v.clone()))
            .collect();
        Ok(certs)
    }

    fn load_private_key(&self) -> Result<rustls::PrivateKey, Box<dyn Error>> {
        let key_file = File::open(&self.key_path)?;
        let mut reader = BufReader::new(key_file);
        let keys = rustls_pemfile::pkcs8_private_keys(&mut reader)?;
        Ok(rustls::PrivateKey(keys[0].clone()))
    }
}
```

```rust
use rcgen::{Certificate, CertificateParams, DnType, PKCS_ECDSA_P256_SHA256};

struct CertificateManager {
    storage_path: String,

impl CertificateManager {
    async fn generate_self_signed_cert(&self, domain: &str) -> Result<(), Box<dyn Error>> {
        let mut params = CertificateParams::new(vec![domain.to_string()]);
```

## 3. 보안 프로토콜 구현

### 3.1 보안 통신 프로토콜

```rust
use serde::{Serialize, Deserialize};
use tokio::io::{AsyncReadExt, AsyncWriteExt};

#[derive(Serialize, Deserialize)]
struct SecureMessage {
    version: u8,
    message_type: MessageType,
    payload: Vec<u8>,
    signature: Vec<u8>,
}

#[derive(Serialize, Deserialize)]
enum MessageType {
    Handshake,
    Data,
    Control,
}

struct SecureProtocol {
    stream: tokio_rustls::client::TlsStream<TcpStream>,
}

impl SecureProtocol {
    async fn send_message(&mut self, message: SecureMessage) -> Result<(), Box<dyn Error>> {
        let data = bincode::serialize(&message)?;
        let len = data.len() as u32;
        self.stream.write_u32(len).await?;
        self.stream.write_all(&data).await?;
        self.stream.flush().await?;
        Ok(())
    }

    async fn receive_message(&mut self) -> Result<SecureMessage, Box<dyn Error>> {
        let len = self.stream.read_u32().await? as usize;
        let mut buffer = vec![0u8; len];
        self.stream.read_exact(&mut buffer).await?;
        let message = bincode::deserialize(&buffer)?;
        Ok(message)
    }
}
```

## 4. 암호화 통신

### 4.1 암호화 구현

```rust
use aes_gcm::{Aes256Gcm, Key, Nonce};
use aes_gcm::aead::{Aead, NewAead};
use rand::Rng;

struct Encryption {
    cipher: Aes256Gcm,
}

impl Encryption {
    fn new(key: &[u8]) -> Result<Self, Box<dyn Error>> {
        let cipher_key = Key::from_slice(key);
        let cipher = Aes256Gcm::new(cipher_key);

        Ok(Encryption { cipher })
    }

    fn encrypt(&self, data: &[u8]) -> Result<Vec<u8>, Box<dyn Error>> {
        let mut rng = rand::thread_rng();
        let mut nonce_bytes = [0u8; 12];
        rng.fill(&mut nonce_bytes);
        let nonce = Nonce::from_slice(&nonce_bytes);

        let encrypted = self.cipher
            .encrypt(nonce, data)
            .map_err(|e| format!("Encryption failed: {}", e))?;

        // Nonce를 암호문과 함께 반환
        let mut result = Vec::with_capacity(nonce_bytes.len() + encrypted.len());
        result.extend_from_slice(&nonce_bytes);
        result.extend_from_slice(&encrypted);

        Ok(result)
    }

    fn decrypt(&self, data: &[u8]) -> Result<Vec<u8>, Box<dyn Error>> {
        if data.len() < 12 {
            return Err("Data too short".into());
        }

        let (nonce_bytes, encrypted) = data.split_at(12);
        let nonce = Nonce::from_slice(nonce_bytes);

        let decrypted = self.cipher
            .decrypt(nonce, encrypted)
            .map_err(|e| format!("Decryption failed: {}", e))?;

        Ok(decrypted)
    }
}
```

이 장에서는 Tokio를 사용한 보안과 암호화 구현의 핵심 내용을 다루었다.

1. TLS 구현
   - 안전한 클라이언트/서버 통신
   - 인증서 관리
   - 설정 최적화

2. 인증서 관리
   - 자체 서명 인증서 생성
   - 인증서 갱신 및 검증
   - 만료 관리

3. 보안 프로토콜
   - 메시지 형식 정의
   - 안전한 통신 구현
   - 프로토콜 버전 관리

4. 암호화 통신
   - AES-GCM 암호화
   - 키 관리
   - 안전한 데이터 전송

5. 보안 모범 사례
   - 자동화된 보안 감사
   - 인증서 순환
   - 안전한 종료 처리

구현 시 주의사항은 다음과 같다.

- 최신 TLS 버전 사용 (TLS 1.3 권장)
- 강력한 암호화 알고리즘 선택
- 정기적인 인증서 갱신
- 안전한 키 관리
- 보안 감사 자동화

이러한 요소들을 적절히 구현하면 안전하고 신뢰성 있는 네트워크 애플리케이션을 구축할 수 있다.
