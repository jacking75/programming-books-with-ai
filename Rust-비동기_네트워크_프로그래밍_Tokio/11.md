```rust
use tracing::{info, warn, error, Level};
use tracing_subscriber::{FmtSubscriber, EnvFilter};

fn setup_logging() {
    // 기본 로그 설정
    let subscriber = FmtSubscriber::builder()
        .with_env_filter(EnvFilter::from_default_env()
            .add_directive(Level::INFO.into())
            .add_directive("my_app=debug".parse().unwrap()))
        .with_thread_ids(true)
        .with_thread_names(true)
        .with_file(true)
        .with_line_number(true)
        .with_target(false)
        .pretty()
        .init();
}

// 로그 사용 예제
#[tracing::instrument]
async fn process_request(user_id: String) -> Result<(), Error> {
    info!(user.id = %user_id, "시작: 요청 처리");

    if user_id.is_empty() {
        error!("잘못된 사용자 ID");
        return Err(Error::InvalidUserId);
    }

    warn!("경고가 필요한 상황 발생");

    info!("완료: 요청 처리");
    Ok(())
}
```

## 2. 메트릭 수집

### 2.1 Prometheus 메트릭 설정

```rust
use prometheus::{
    Registry, Counter, Gauge, Histogram,
    HistogramOpts, Opts,
};

struct Metrics {
    requests_total: Counter,
    active_connections: Gauge,
    request_duration: Histogram,
    registry: Registry,
}

impl Metrics {
    fn new() -> Self {
        let registry = Registry::new();

        let requests_total = Counter::new(
            "http_requests_total",
            "총 HTTP 요청 수"
        ).unwrap();

        let active_connections = Gauge::new(
            "active_connections",
            "현재 활성 연결 수"
        ).unwrap();

        let request_duration = Histogram::with_opts(
            HistogramOpts::new(
                "http_request_duration_seconds",
                "HTTP 요청 처리 시간"
            )
        ).unwrap();

        registry.register(Box::new(requests_total.clone())).unwrap();
        registry.register(Box::new(active_connections.clone())).unwrap();
        registry.register(Box::new(request_duration.clone())).unwrap();

        Metrics {
            requests_total,
            active_connections,
            request_duration,
            registry,
        }
    }
}
```

### 2.2 메트릭 수집 미들웨어

```rust
use axum::{
    middleware::{Next, Handler},
    response::Response,
};

#[derive(Clone)]
struct MetricsMiddleware {
    metrics: Arc<Metrics>,
}

impl MetricsMiddleware {
    async fn handle<B>(
        &self,
        request: Request<B>,
        next: Next<B>,
    ) -> Response {
        let start = Instant::now();
        self.metrics.active_connections.inc();

        let response = next.run(request).await;

        self.metrics.active_connections.dec();
        self.metrics.requests_total.inc();
        self.metrics.request_duration
            .observe(start.elapsed().as_secs_f64());

        response
    }
}
```
  

## 3. 성능 모니터링

### 3.1 시스템 메트릭 수집

```rust
use sysinfo::{System, SystemExt, ProcessExt};

struct SystemMetrics {
    sys: System,
    metrics: Arc<Metrics>,
}

impl SystemMetrics {
    fn new(metrics: Arc<Metrics>) -> Self {
        SystemMetrics {
            sys: System::new_all(),
            metrics,
        }
    }

    async fn collect_metrics(&mut self) {
        loop {
            self.sys.refresh_all();

            // CPU 사용량
            let cpu_usage = self.sys.global_cpu_info().cpu_usage();
            self.metrics.cpu_usage.set(cpu_usage);

            // 메모리 사용량
            let memory_usage = self.sys.used_memory() as f64;
            self.metrics.memory_usage.set(memory_usage);

            // 프로세스 정보
            if let Some(process) = self.sys.process(std::process::id() as i32) {
                self.metrics.process_cpu_usage
                    .set(process.cpu_usage() as f64);
                self.metrics.process_memory
                    .set(process.memory() as f64);
            }

            sleep(Duration::from_secs(15)).await;
        }
    }
}
```

### 3.2 성능 프로파일링

```rust
use tokio::time::Instant;
use std::collections::HashMap;

struct Profiler {
    start_times: HashMap<String, Instant>,
    durations: HashMap<String, Vec<Duration>>,
}

impl Profiler {
    fn new() -> Self {
        Profiler {
            start_times: HashMap::new(),
            durations: HashMap::new(),
        }
    }

    fn start(&mut self, operation: &str) {
        self.start_times.insert(
            operation.to_string(),
            Instant::now()
        );
    }

    fn end(&mut self, operation: &str) {
        if let Some(start_time) = self.start_times.remove(operation) {
            let duration = start_time.elapsed();
            self.durations
                .entry(operation.to_string())
                .or_insert_with(Vec::new)
                .push(duration);
        }
    }

    fn report(&self) -> HashMap<String, Statistics> {
        let mut stats = HashMap::new();

        for (op, durations) in &self.durations {
            let count = durations.len();
            let total: Duration = durations.iter().sum();
            let avg = total / count as u32;

            stats.insert(op.clone(), Statistics {
                count,
                total,
                average: avg,
            });
        }

        stats
    }
}
```
  

## 4. 트레이싱

### 4.1 OpenTelemetry 설정

```rust
use opentelemetry::{
    trace::{Tracer, TracerProvider},
    sdk::{trace::Config, Resource},
};

fn setup_tracing() -> Result<(), Error> {
    // OpenTelemetry 트레이서 설정
    let tracer = opentelemetry_jaeger::new_pipeline()
        .with_service_name("my-service")
        .with_collector_endpoint("http://jaeger:14268/api/traces")
        .install_batch(opentelemetry::runtime::Tokio)?;

    // Tracing 서브스크라이버 설정
    tracing_subscriber::registry()
        .with(tracing_subscriber::EnvFilter::new("INFO"))
        .with(tracing_opentelemetry::layer().with_tracer(tracer))
        .try_init()?;

    Ok(())
}
```

### 4.2 분산 트레이싱

```rust
use opentelemetry::trace::TraceContextExt;
use tracing_opentelemetry::OpenTelemetrySpanExt;

#[tracing::instrument(fields(trace_id))]
async fn handle_request(req: Request) -> Result<Response, Error> {
    let trace_id = req.headers()
        .get("x-trace-id")
        .map(|h| h.to_str().unwrap_or_default())
        .unwrap_or_default();

    Span::current().set_attribute("trace_id", trace_id.to_string());

    let db_span = tracing::info_span!("database_operation");
    let db_result = db_span.in_scope(|| async {
        // 데이터베이스 작업
        Ok(())
    }).await?;

    let processing_span = tracing::info_span!("processing");
    let result = processing_span.in_scope(|| async {
        // 비즈니스 로직 처리
        Ok(Response::new())
    }).await?;

    Ok(result)
}
```
  

## 5. 오류 추적

### 5.1 오류 처리와 보고

```rust
use sentry::{capture_error, configure_scope};

#[derive(Debug, thiserror::Error)]
enum AppError {
    #[error("데이터베이스 오류: {0}")]
    Database(#[from] sqlx::Error),

    #[error("유효하지 않은 입력: {0}")]
    InvalidInput(String),

    #[error("내부 서버 오류: {0}")]
    Internal(String),
}

async fn handle_error(error: AppError) {
    // 오류 로깅
    error!(error = %error, "오류 발생");

    // Sentry에 오류 보고
    with_scope(|scope| {
        scope.set_tag("error_type", error.type_name());
        capture_error(&error);
    });

    // 메트릭 증가
    ERRORS_TOTAL
        .with_label_values(&[error.type_name()])
        .inc();
}
```

### 5.2 오류 모니터링 시스템

```rust
struct ErrorMonitor {
    error_counts: HashMap<String, AtomicUsize>,
    alert_threshold: usize,
    notification_sender: mpsc::Sender<AlertMessage>,
}

impl ErrorMonitor {
    async fn monitor_errors(&self) {
        loop {
            for (error_type, count) in &self.error_counts {
                let current_count = count.load(Ordering::SeqCst);

                if current_count >= self.alert_threshold {
                    self.send_alert(error_type, current_count).await;
                    count.store(0, Ordering::SeqCst);
                }
            }

            sleep(Duration::from_secs(60)).await;
        }
    }

    async fn send_alert(&self, error_type: &str, count: usize) {
        let alert = AlertMessage {
            error_type: error_type.to_string(),
            count,
            timestamp: Utc::now(),
        };

        if let Err(e) = self.notification_sender.send(alert).await {
            error!("알림 전송 실패: {}", e);
        }
    }
}
```
