## 1. 채팅 서버 구현

### 1.1 기본 채팅 서버 구조

```rust
use tokio::{
    net::{TcpListener, TcpStream},
    sync::broadcast,
    io::{AsyncReadExt, AsyncWriteExt},
};

#[derive(Clone, Debug)]
struct ChatMessage {
    from: String,
    content: String,
    timestamp: DateTime<Utc>,
}

struct ChatServer {
    tx: broadcast::Sender<ChatMessage>,
    clients: Arc<Mutex<HashMap<String, Client>>>,
}

impl ChatServer {
    fn new() -> Self {
        let (tx, _) = broadcast::channel(100);
        ChatServer {
            tx,
            clients: Arc::new(Mutex::new(HashMap::new())),
        }
    }

    async fn run(&self, addr: &str) -> Result<(), Box<dyn Error>> {
        let listener = TcpListener::bind(addr).await?;
        println!("채팅 서버 시작: {}", addr);

        loop {
            let (socket, addr) = listener.accept().await?;
            let tx = self.tx.clone();
            let clients = self.clients.clone();

            tokio::spawn(async move {
                if let Err(e) = handle_connection(socket, addr, tx, clients).await {
                    eprintln!("클라이언트 처리 오류: {}", e);
                }
            });
        }
    }
}
```

### 1.2 클라이언트 핸들링

```rust
async fn handle_connection(
    mut socket: TcpStream,
    addr: SocketAddr,
    tx: broadcast::Sender<ChatMessage>,
    clients: Arc<Mutex<HashMap<String, Client>>>,
) -> Result<(), Box<dyn Error>> {
    // 사용자 이름 읽기
    let mut username = String::new();
    socket.read_line(&mut username).await?;
    let username = username.trim().to_string();

    // 클라이언트 등록
    let client = Client {
        addr,
        username: username.clone(),
        socket: socket.clone(),
    };
    clients.lock().await.insert(username.clone(), client);

    // 메시지 수신 채널 구독
    let mut rx = tx.subscribe();

    loop {
        tokio::select! {
            // 클라이언트로부터 메시지 수신
            result = socket.read_line(&mut String::new()) => {
                match result {
                    Ok(0) => break, // 연결 종료
                    Ok(n) => {
                        let msg = ChatMessage {
                            from: username.clone(),
                            content: message,
                            timestamp: Utc::now(),
                        };
                        tx.send(msg)?;
                    }
                    Err(e) => return Err(e.into()),
                }
            }
            // 브로드캐스트 메시지 수신
            result = rx.recv() => {
                match result {
                    Ok(msg) => {
                        let formatted = format!(
                            "[{}] {}: {}\n",
                            msg.timestamp,
                            msg.from,
                            msg.content
                        );
                        socket.write_all(formatted.as_bytes()).await?;
                    }
                    Err(e) => return Err(e.into()),
                }
            }
        }
    }

    // 클라이언트 연결 종료 처리
    clients.lock().await.remove(&username);
    Ok(())
}
```

### 2.2 HTTPS 서버 구현

```rust
use tokio_rustls::rustls::{
    ServerConfig,
    AllowAnyAuthenticatedClient,
    NoClientAuth,
    internal::pemfile,
};

async fn create_https_server() -> Result<(), Box<dyn Error>> {
    // TLS 설정
    let cert_file = File::open("cert.pem")?;
    let key_file = File::open("key.pem")?;

    let cert_chain = pemfile::certs(&mut BufReader::new(cert_file))?;
    let key = pemfile::pkcs8_private_keys(&mut BufReader::new(key_file))?[0].clone();

    let config = ServerConfig::builder()
        .with_safe_defaults()
        .with_no_client_auth()
        .with_single_cert(cert_chain, key)?;

    let tls_acceptor = TlsAcceptor::from(Arc::new(config));

    // HTTPS 서버 시작
    let addr = SocketAddr::from(([127, 0, 0, 1], 443));
    let listener = TcpListener::bind(addr).await?;

    println!("HTTPS 서버 시작: {}", addr);

    while let Ok((stream, _)) = listener.accept().await {
        let acceptor = tls_acceptor.clone();

        tokio::spawn(async move {
            match acceptor.accept(stream).await {
                Ok(tls_stream) => {
                    // TLS 스트림 처리
                    handle_https_connection(tls_stream).await;
                }
                Err(e) => eprintln!("TLS 핸드쉐이크 실패: {}", e),
            }
        });
    }

    Ok(())
}
```

## 3. 웹소켓 서버

```rust
use tokio_tungstenite::{
    accept_async,
    tungstenite::Message,
};

struct WebSocketServer {
    clients: Arc<Mutex<HashMap<usize, WebSocketSender>>>,
    next_client_id: AtomicUsize,
}

impl WebSocketServer {
    async fn run(&self, addr: &str) -> Result<(), Box<dyn Error>> {
        let listener = TcpListener::bind(addr).await?;
        println!("웹소켓 서버 시작: {}", addr);

        while let Ok((stream, _)) = listener.accept().await {
            let ws_stream = accept_async(stream).await?;
            let client_id = self.next_client_id.fetch_add(1, Ordering::Relaxed);
            let clients = self.clients.clone();

            tokio::spawn(async move {
                if let Err(e) = handle_websocket_client(
                    ws_stream,
                    client_id,
                    clients,
                ).await {
                    eprintln!("웹소켓 클라이언트 처리 오류: {}", e);
                }
            });
        }

        Ok(())
    }
}

async fn handle_websocket_client(
    mut ws_stream: WebSocketStream<TcpStream>,
    client_id: usize,
    clients: Arc<Mutex<HashMap<usize, WebSocketSender>>>,
) -> Result<(), Box<dyn Error>> {
    while let Some(msg) = ws_stream.next().await {
        match msg? {
            Message::Text(text) => {
                // 모든 클라이언트에게 메시지 브로드캐스트
                let clients = clients.lock().await;
                for (_, sender) in clients.iter() {
                    sender.send(Message::Text(text.clone())).await?;
                }
            }
            Message::Close(_) => break,
            _ => {}
        }
    }

    // 클라이언트 연결 종료 처리
    clients.lock().await.remove(&client_id);
    Ok(())
}
```
  

## 4. 프록시 서버

```rust
use tokio::io::{copy_bidirectional};

struct ProxyServer {
    upstream: String,
    max_connections: usize,
}

impl ProxyServer {
    async fn run(&self, addr: &str) -> Result<(), Box<dyn Error>> {
        let listener = TcpListener::bind(addr).await?;
        let semaphore = Arc::new(Semaphore::new(self.max_connections));

        println!("프록시 서버 시작: {}", addr);

        while let Ok((inbound, _)) = listener.accept().await {
            let permit = semaphore.clone().acquire_owned().await?;
            let upstream = self.upstream.clone();

            tokio::spawn(async move {
                if let Err(e) = handle_proxy_connection(inbound, &upstream).await {
                    eprintln!("프록시 연결 오류: {}", e);
                }
                drop(permit);
            });
        }

        Ok(())
    }
}

async fn handle_proxy_connection(
    mut inbound: TcpStream,
    upstream: &str,
) -> Result<(), Box<dyn Error>> {
    let mut outbound = TcpStream::connect(upstream).await?;

    // 양방향 데이터 복사
    let (mut ri, mut wi) = inbound.split();
    let (mut ro, mut wo) = outbound.split();

    let client_to_server = copy(&mut ri, &mut wo);
    let server_to_client = copy(&mut ro, &mut wi);

    try_join!(client_to_server, server_to_client)?;

    Ok(())
}
```
  

## 5. 로드 밸런서 구현

```rust
use rand::seq::SliceRandom;

#[derive(Clone)]
struct Backend {
    addr: String,
    healthy: Arc<AtomicBool>,
    last_check: Arc<Mutex<Instant>>,
}

struct LoadBalancer {
    backends: Vec<Backend>,
    strategy: LoadBalancingStrategy,
    health_check_interval: Duration,
}

enum LoadBalancingStrategy {
    RoundRobin(Arc<AtomicUsize>),
    Random,
    LeastConnections(Arc<Mutex<HashMap<String, usize>>>),
}

impl LoadBalancer {
    async fn run(&self, addr: &str) -> Result<(), Box<dyn Error>> {
        // 상태 체크 시작
        self.start_health_checker();

        let listener = TcpListener::bind(addr).await?;
        println!("로드 밸런서 시작: {}", addr);

        while let Ok((inbound, _)) = listener.accept().await {
            let backend = self.choose_backend().await?;

            tokio::spawn(async move {
                if let Err(e) = forward_connection(inbound, &backend).await {
                    eprintln!("연결 전달 오류: {}", e);
                }
            });
        }

        Ok(())
    }

    async fn choose_backend(&self) -> Result<Backend, Box<dyn Error>> {
        let healthy_backends: Vec<_> = self.backends.iter()
            .filter(|b| b.healthy.load(Ordering::Relaxed))
            .collect();

        if healthy_backends.is_empty() {
            return Err("사용 가능한 백엔드 없음".into());
        }

        match &self.strategy {
            LoadBalancingStrategy::RoundRobin(counter) => {
                let idx = counter.fetch_add(1, Ordering::Relaxed) % healthy_backends.len();
                Ok(healthy_backends[idx].clone())
            }
            LoadBalancingStrategy::Random => {
                let backend = healthy_backends.choose(&mut rand::thread_rng())
                    .ok_or("백엔드 선택 실패")?;
                Ok(backend.clone())
            }
            LoadBalancingStrategy::LeastConnections(connections) => {
                let conns = connections.lock().await;
                let backend = healthy_backends.iter()
                    .min_by_key(|b| conns.get(&b.addr).unwrap_or(&0))
                    .ok_or("백엔드 선택 실패")?;
                Ok(backend.clone())
            }
        }
    }

    fn start_health_checker(&self) {
        let backends = self.backends.clone();
        let interval = self.health_check_interval;

        tokio::spawn(async move {
            loop {
                for backend in &backends {
                    let healthy = check_backend_health(&backend.addr).await;
                    backend.healthy.store(healthy, Ordering::Relaxed);
                    *backend.last_check.lock().await = Instant::now();
                }
                sleep(interval).await;
            }
        });
    }
}
```
