### 1.2 Tokio 특화 테스트 유틸리티

```rust
use tokio::time::{sleep, timeout, Duration};

#[cfg(test)]
mod async_tests {
    use super::*;

    // 타임아웃 테스트
    #[tokio::test]
    async fn test_with_timeout() {
        let result = timeout(
            Duration::from_secs(1),
            async_operation()
        ).await;

        assert!(result.is_ok());
    }

    // 병렬 테스트
    #[tokio::test(flavor = "multi_thread", worker_threads = 2)]
    async fn test_concurrent_operations() {
        let (tx, rx) = tokio::sync::mpsc::channel(1);

        let send_task = tokio::spawn(async move {
            tx.send("test").await.unwrap();
        });

        let receive_task = tokio::spawn(async move {
            assert_eq!(rx.recv().await.unwrap(), "test");
        });

        tokio::try_join!(send_task, receive_task).unwrap();
    }
}
```

## 4. 비동기 코드 테스트

### 4.1 비동기 테스트 패턴

```rust
#[tokio::test]
async fn test_async_stream() {
    use tokio_stream::{self as stream, StreamExt};

    let mut stream = stream::iter(vec![1, 2, 3, 4, 5]);

    let mut sum = 0;
    while let Some(num) = stream.next().await {
        sum += num;
    }

    assert_eq!(sum, 15);
}

#[tokio::test]
async fn test_concurrent_operations() {
    let mut handles = vec![];

    for i in 0..3 {
        let handle = tokio::spawn(async move {
            // 비동기 작업
            Ok::<_, Error>(i)
        });
        handles.push(handle);
    }

    let results = futures::future::join_all(handles).await;
    assert!(results.iter().all(|r| r.is_ok()));
}
```

### 4.2 비동기 테스트 유틸리티

```rust
struct AsyncTestUtils;

impl AsyncTestUtils {
    async fn wait_for_condition<F>(
        condition: F,
        timeout: Duration,
    ) -> Result<(), Error>
    where
        F: Fn() -> bool,
    {
        let start = Instant::now();
        while !condition() {
            if start.elapsed() > timeout {
                return Err(Error::Timeout);
            }
            sleep(Duration::from_millis(100)).await;
        }
        Ok(())
    }

    async fn with_timeout<F, T>(
        future: F,
        duration: Duration,
    ) -> Result<T, Error>
    where
        F: Future<Output = Result<T, Error>>,
    {
        match timeout(duration, future).await {
            Ok(result) => result,
            Err(_) => Err(Error::Timeout),
        }
    }
}
```

async fn process_request(req: Request) -> Result<Response, Error> {
    info!("Processing request: {:?}", req);

### 5.2 디버깅 도구

```rust
use std::sync::atomic::{AtomicUsize, Ordering};

struct DebugCounter {
    count: AtomicUsize,
}

impl DebugCounter {
    fn new() -> Self {
        DebugCounter {
            count: AtomicUsize::new(0),
        }
    }

    fn increment(&self) {
        self.count.fetch_add(1, Ordering::SeqCst);
    }

    fn get_count(&self) -> usize {
        self.count.load(Ordering::SeqCst)
    }
}

#[tokio::test]
async fn debug_async_flow() {
    let counter = Arc::new(DebugCounter::new());
    let mut handles = vec![];

    for _ in 0..10 {
        let counter = counter.clone();
        handles.push(tokio::spawn(async move {
            counter.increment();
        }));
    }

    futures::future::join_all(handles).await;
    assert_eq!(counter.get_count(), 10);
}
```
