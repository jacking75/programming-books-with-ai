CHAPTER 07

간단한 에코 서버 구현

1장의 블로킹 에코 서버, 3장의 세션 기반 RIO 서버, 6장의 래퍼 클래스 기반 에코 서버에 이어, 이번 장은 지금까지 배운 모든 것을 하나의 완결된 파일로 처음부터 다시 쌓아 올리는 "정리판"입니다. 클래스를 선언하고, 초기화·연결 수락·송수신·완료 처리를 각각 구현하고, 마지막에 실전에서 자주 만나는 RIO 오류들과 그 대처법까지 정리합니다. 8장부터는 이 골격 위에 실제 게임 패킷을 얹기 시작하므로, 이 장에서 구조를 확실히 다져둡니다.

이 장에서 배우는 것

  • 선언과 구현을 분리해 RIO 서버 클래스를 처음부터 끝까지 완성하는 흐름
  • FormatMessageA()로 Winsock 오류 코드를 사람이 읽을 문장으로 바꾸는 법
  • 완료가 수신인지 송신인지 상태 플래그로 구분하는 방식과 그 한계
  • std::expected + try/catch를 함께 쓰는 예외 안전 설계
  • 실전에서 자주 마주치는 RIO 오류 6가지와 원인·해결책

7.1 RIO를 이용한 기본적인 에코 서버

지금까지 배운 RIO의 네 가지 특징을 다시 한 번 짚고, 클래스 골격부터 세웁니다.

  1. 사전 등록된 버퍼 — 모든 송수신 버퍼는 미리 RIORegisterBuffer()로 등록되어야 합니다(2·4장).
  2. 비동기 완료 통지 — 이벤트 기반 또는 폴링 기반으로 완료를 통지받습니다(5장).
  3. 요청 큐와 완료 큐 — 비동기 작업은 큐 두 종류로 관리됩니다(2·5장).
  4. 컨텍스트 기반 처리 — 요청마다 컨텍스트(주로 클라이언트 포인터)를 붙여 완료 시 "누구의 요청이었는지" 식별합니다(3장부터 계속).

아래 클래스 선언은 지금까지 만든 여러 버전의 RIOEchoServer·RIOServer와 구조가 비슷합니다. 다른 점은 클라이언트 상태를 is_receiving/ is_sending 두 개의 불(bool) 플래그로 추적한다는 것인데, 이 설계의 장단점은 7.3절에서 자세히 다룹니다.

#pragma comment(lib, "ws2_32.lib")
#pragma comment(lib, "mswsock.lib")

GUID g_RIOFunctionTableId = WSAID_MULTIPLE_RIO;

class RIOEchoServer {
private:
    RIO_EXTENSION_FUNCTION_TABLE rio_fn_table_{};
    SOCKET listen_socket_ = INVALID_SOCKET;
    RIO_CQ completion_queue_ = RIO_INVALID_CQ;
    RIO_BUFFERID send_buffer_id_ = RIO_INVALID_BUFFERID;
    RIO_BUFFERID recv_buffer_id_ = RIO_INVALID_BUFFERID;

    std::vector<char> send_buffer_;
    std::vector<char> recv_buffer_;
    size_t buffer_size_ = 4096;
    size_t total_buffer_size_ = 1024 * 1024;

    uint16_t port_ = 9000;
    int max_connections_ = 1000;
    std::atomic<bool> running_ = false;
    std::atomic<size_t> outstanding_requests_ = 0;
    bool winsock_initialized_ = false;

    std::jthread worker_thread_;
    std::jthread accept_thread_;

    HANDLE completion_event_ = NULL;
    bool use_event_notification_ = true;

    struct ClientContext {
        SOCKET socket = INVALID_SOCKET;
        RIO_RQ request_queue = RIO_INVALID_RQ;
        size_t buffer_offset = 0;
        size_t send_offset = 0;
        size_t send_remaining = 0;
        bool is_receiving = false;
        bool is_sending = false;
    };

    std::vector<std::unique_ptr<ClientContext>> clients_;
    std::vector<size_t> free_buffer_offsets_;
    std::mutex clients_mutex_;

public:
    RIOEchoServer(uint16_t port = 9000, size_t buffer_size = 4096,
                 bool use_event_notification = true);
    ~RIOEchoServer();

    std::expected<void, std::string> Initialize();
    std::expected<void, std::string> Start();
    void Stop();

private:
    std::expected<void, std::string> InitializeWinsock();
    std::expected<void, std::string> InitializeRIOFunctionTable();
    std::expected<void, std::string> CreateSocket();
    std::expected<void, std::string> BindAndListen();
    std::expected<void, std::string> RegisterBuffers();
    std::expected<void, std::string> CreateCompletionQueue();

    void AcceptClients(std::stop_token stop_token);
    std::expected<void, std::string> HandleNewClient(SOCKET client_socket);
    void RemoveClient(ClientContext* client);

    std::expected<void, std::string> PostReceive(ClientContext* client);
    std::expected<void, std::string> PostSend(ClientContext* client, size_t data_size, size_t data_offset = 0);

    void ProcessCompletions(std::stop_token stop_token);
    void HandleCompletion(RIORESULT* results, ULONG result_count);

    std::string GetErrorMessage(DWORD error_code) const;
    std::string GetLastErrorMessage() const;
    void LogInfo(std::string_view message);
    void LogError(std::string_view message);
};

7.2 클라이언트 연결 처리

서버 초기화

Initialize()는 지금까지 여러 장에서 반복한 순서 — Winsock 초기화 → 함수 테이블 → 소켓 생성 → 버퍼 등록 → 완료 큐 → 바인딩/리스닝 — 을 그대로 따릅니다. 각 단계가 std::expected를 반환하므로, 어느 하나라도 실패하면 그 자리에서 바로 오류를 전파합니다. 

RIOEchoServer::RIOEchoServer(uint16_t port, size_t buffer_size, bool use_event_notification)
    : port_(port), buffer_size_(buffer_size), use_event_notification_(use_event_notification) {
    if (buffer_size == 0 ||
        buffer_size > (std::numeric_limits<DWORD>::max)() / static_cast<size_t>(max_connections_)) {
        throw std::invalid_argument("클라이언트당 버퍼 크기가 유효한 RIO 버퍼 범위를 벗어났습니다");
    }

    total_buffer_size_ = buffer_size * max_connections_;
    send_buffer_.resize(total_buffer_size_);
    recv_buffer_.resize(total_buffer_size_);

    free_buffer_offsets_.reserve(max_connections_);
    for (size_t i = static_cast<size_t>(max_connections_); i > 0; --i) {
        free_buffer_offsets_.push_back((i - 1) * buffer_size_);
    }
}

RIOEchoServer::~RIOEchoServer() { Stop(); }

std::expected<void, std::string> RIOEchoServer::Initialize() {
    auto result = InitializeWinsock();
    if (!result) return result;

    result = InitializeRIOFunctionTable();
    if (!result) { Stop(); return result; }

    result = CreateSocket();
    if (!result) { Stop(); return result; }

    result = RegisterBuffers();
    if (!result) { Stop(); return result; }

    result = CreateCompletionQueue();
    if (!result) { Stop(); return result; }

    result = BindAndListen();
    if (!result) { Stop(); return result; }

    std::print("서버 초기화 완료 - 포트: {}\n", port_);
    return {};
}

std::expected<void, std::string> RIOEchoServer::InitializeWinsock() {
    WSADATA wsa_data;
    const int error_code = WSAStartup(MAKEWORD(2, 2), &wsa_data);
    if (error_code != 0) return std::unexpected(GetErrorMessage(static_cast<DWORD>(error_code)));
    winsock_initialized_ = true;
    return {};
}

std::expected<void, std::string> RIOEchoServer::InitializeRIOFunctionTable() {
    SOCKET temp_socket = WSASocketW(AF_INET, SOCK_STREAM, IPPROTO_TCP, NULL, 0, WSA_FLAG_REGISTERED_IO);
    if (temp_socket == INVALID_SOCKET) return std::unexpected(GetLastErrorMessage());

    DWORD bytes = 0;
    int result = WSAIoctl(
        temp_socket, SIO_GET_MULTIPLE_EXTENSION_FUNCTION_POINTER,
        &g_RIOFunctionTableId, sizeof(g_RIOFunctionTableId),
        &rio_fn_table_, sizeof(rio_fn_table_),
        &bytes, NULL, NULL
    );
    closesocket(temp_socket);

    if (result == SOCKET_ERROR) return std::unexpected(GetLastErrorMessage());
    return {};
}

std::expected<void, std::string> RIOEchoServer::CreateSocket() {
    listen_socket_ = WSASocketW(AF_INET, SOCK_STREAM, IPPROTO_TCP, NULL, 0, WSA_FLAG_REGISTERED_IO);
    if (listen_socket_ == INVALID_SOCKET) return std::unexpected(GetLastErrorMessage());

    int opt_value = 1;
    if (setsockopt(listen_socket_, SOL_SOCKET, SO_REUSEADDR,
                  reinterpret_cast<char*>(&opt_value), sizeof(opt_value)) == SOCKET_ERROR) {
        return std::unexpected(GetLastErrorMessage());
    }
    return {};
}

std::expected<void, std::string> RIOEchoServer::BindAndListen() {
    sockaddr_in server_addr;
    server_addr.sin_family = AF_INET;
    server_addr.sin_addr.s_addr = htonl(INADDR_ANY);
    server_addr.sin_port = htons(port_);

    if (bind(listen_socket_, reinterpret_cast<sockaddr*>(&server_addr), sizeof(server_addr)) == SOCKET_ERROR) {
        return std::unexpected(GetLastErrorMessage());
    }
    if (listen(listen_socket_, SOMAXCONN) == SOCKET_ERROR) return std::unexpected(GetLastErrorMessage());
    return {};
}

std::expected<void, std::string> RIOEchoServer::RegisterBuffers() {
    send_buffer_id_ = rio_fn_table_.RIORegisterBuffer(send_buffer_.data(), static_cast<DWORD>(send_buffer_.size()));
    if (send_buffer_id_ == RIO_INVALID_BUFFERID) return std::unexpected(GetLastErrorMessage());

    recv_buffer_id_ = rio_fn_table_.RIORegisterBuffer(recv_buffer_.data(), static_cast<DWORD>(recv_buffer_.size()));
    if (recv_buffer_id_ == RIO_INVALID_BUFFERID) return std::unexpected(GetLastErrorMessage());
    return {};
}

std::expected<void, std::string> RIOEchoServer::CreateCompletionQueue() {
    if (use_event_notification_) {
        completion_event_ = CreateEvent(NULL, FALSE, FALSE, NULL);
        if (completion_event_ == NULL) return std::unexpected(GetLastErrorMessage());

        RIO_NOTIFICATION_COMPLETION completion_type{};
        completion_type.Type = RIO_EVENT_COMPLETION;
        completion_type.Event.EventHandle = completion_event_;
        completion_type.Event.NotifyReset = TRUE;

        completion_queue_ = rio_fn_table_.RIOCreateCompletionQueue(max_connections_ * 2, &completion_type);
    } else {
        completion_queue_ = rio_fn_table_.RIOCreateCompletionQueue(max_connections_ * 2, NULL);
    }

    if (completion_queue_ == RIO_INVALID_CQ) return std::unexpected(GetLastErrorMessage());
    return {};
}

std::expected<void, std::string> RIOEchoServer::Start() {
    if (running_.exchange(true)) return std::unexpected("서버가 이미 실행 중입니다");
    if (listen_socket_ == INVALID_SOCKET || completion_queue_ == RIO_INVALID_CQ) {
        running_ = false;
        return std::unexpected("Initialize()를 성공적으로 호출한 뒤에 Start()를 호출해야 합니다");
    }

    worker_thread_ = std::jthread([this](std::stop_token stop_token) { this->ProcessCompletions(stop_token); });
    accept_thread_ = std::jthread([this](std::stop_token stop_token) { this->AcceptClients(stop_token); });

    std::print("에코 서버 시작됨 - 포트: {}\n", port_);
    return {};
}

void RIOEchoServer::Stop() {
    running_ = false;

    accept_thread_.request_stop();

    if (listen_socket_ != INVALID_SOCKET) {
        shutdown(listen_socket_, SD_BOTH);
        closesocket(listen_socket_);
        listen_socket_ = INVALID_SOCKET;
    }
    if (accept_thread_.joinable()) accept_thread_.join();

    {
        std::lock_guard<std::mutex> lock(clients_mutex_);
        for (const auto& client : clients_) {
            if (client->socket != INVALID_SOCKET) {
                closesocket(client->socket); // 미결 RIO 요청을 취소하고 완료를 발생시킨다.
                client->socket = INVALID_SOCKET;
            }
        }
    }

    // 워커는 outstanding_requests_가 0이 될 때까지 취소 완료를 dequeue한다.
    if (use_event_notification_ && completion_event_ != NULL) SetEvent(completion_event_);
    if (worker_thread_.joinable()) worker_thread_.join();

    {
        std::lock_guard<std::mutex> lock(clients_mutex_);
        clients_.clear();
    }

    if (completion_queue_ != RIO_INVALID_CQ) {
        rio_fn_table_.RIOCloseCompletionQueue(completion_queue_);
        completion_queue_ = RIO_INVALID_CQ;
    }
    if (send_buffer_id_ != RIO_INVALID_BUFFERID) {
        rio_fn_table_.RIODeregisterBuffer(send_buffer_id_);
        send_buffer_id_ = RIO_INVALID_BUFFERID;
    }
    if (recv_buffer_id_ != RIO_INVALID_BUFFERID) {
        rio_fn_table_.RIODeregisterBuffer(recv_buffer_id_);
        recv_buffer_id_ = RIO_INVALID_BUFFERID;
    }
    if (completion_event_ != NULL) {
        CloseHandle(completion_event_);
        completion_event_ = NULL;
    }

    if (winsock_initialized_) {
        WSACleanup();
        winsock_initialized_ = false;
    }
    std::print("에코 서버 종료됨\n");
}

클라이언트 연결 수락 및 관리

void RIOEchoServer::AcceptClients(std::stop_token stop_token) {
    while (!stop_token.stop_requested() && running_) {
        fd_set readable{};
        FD_SET(listen_socket_, &readable);
        timeval timeout{ 0, 200000 }; // Stop() 요청을 최대 200ms 안에 확인한다.
        const int ready = select(0, &readable, nullptr, nullptr, &timeout);
        if (ready == 0) continue;
        if (ready == SOCKET_ERROR) {
            if (running_) LogError(std::format("수락 소켓 대기 실패: {}", GetLastErrorMessage()));
            continue;
        }

        sockaddr_in client_addr;
        int addr_len = sizeof(client_addr);

        SOCKET client_socket = accept(listen_socket_, reinterpret_cast<sockaddr*>(&client_addr), &addr_len);
        if (client_socket == INVALID_SOCKET) {
            if (running_) LogError(std::format("클라이언트 연결 수락 실패: {}", GetLastErrorMessage()));
            continue;
        }
        if (!running_) {
            closesocket(client_socket);
            break;
        }

        char client_ip[INET_ADDRSTRLEN];
        inet_ntop(AF_INET, &client_addr.sin_addr, client_ip, sizeof(client_ip));
        LogInfo(std::format("새 클라이언트 연결됨: {}:{}", client_ip, ntohs(client_addr.sin_port)));

        auto result = HandleNewClient(client_socket);
        if (!result) {
            LogError(std::format("클라이언트 처리 실패: {}", result.error()));
        }
    }
}

std::expected<void, std::string> RIOEchoServer::HandleNewClient(SOCKET client_socket) {
    auto client = std::make_unique<ClientContext>();
    client->socket = client_socket;
    ClientContext* client_context = client.get();

    {
        std::lock_guard<std::mutex> lock(clients_mutex_);
        if (free_buffer_offsets_.empty()) {
            closesocket(client_socket);
            return std::unexpected("최대 연결 수 초과");
        }

        client->buffer_offset = free_buffer_offsets_.back();
        free_buffer_offsets_.pop_back();
        client->request_queue = rio_fn_table_.RIOCreateRequestQueue(
            client_socket, 1, 1, 1, 1, completion_queue_, completion_queue_, client.get()
        );
        if (client->request_queue == RIO_INVALID_RQ) {
            free_buffer_offsets_.push_back(client->buffer_offset);
            closesocket(client_socket);
            return std::unexpected(GetLastErrorMessage());
        }

        // 완료가 즉시 도착해도 RequestContext가 유효하도록 먼저 소유권을 등록한다.
        clients_.push_back(std::move(client));

        auto recv_result = PostReceive(client_context);
        if (!recv_result) {
            closesocket(client_context->socket);
            free_buffer_offsets_.push_back(client_context->buffer_offset);
            clients_.pop_back();
            return recv_result;
        }
    }
    return {};
}

void RIOEchoServer::RemoveClient(ClientContext* client) {
    if (!client) return;

    std::lock_guard<std::mutex> lock(clients_mutex_);
    auto it = std::find_if(clients_.begin(), clients_.end(),
                         [client](const auto& c) { return c.get() == client; });
    if (it != clients_.end()) {
        if ((*it)->socket != INVALID_SOCKET) {
            closesocket((*it)->socket);
            (*it)->socket = INVALID_SOCKET;
        }
        free_buffer_offsets_.push_back((*it)->buffer_offset);
        LogInfo("클라이언트 연결 종료");
        clients_.erase(it);
    }
}

최대 동시 요청 수를 1로 제한한 이유

RIOCreateRequestQueue()의 최대 수신·송신 요청 수를 모두 1로 두었습니다. 이 예제는 수신 완료 뒤에만 송신을 요청하고, 송신이 모두 끝난 뒤에만 다음 수신을 요청하므로 실제로는 클라이언트당 미결 요청이 항상 하나입니다. 이 엄격한 순서가 7.3절의 상태 플래그 기반 완료 판별 방식이 성립하는 전제 조건이기도 합니다.

7.3 데이터 송수신 구현

std::expected<void, std::string> RIOEchoServer::PostReceive(ClientContext* client) {
    if (!client || client->request_queue == RIO_INVALID_RQ) {
        return std::unexpected("유효하지 않은 클라이언트");
    }
    if (client->is_receiving) return {};   // 이미 진행 중이면 조용히 넘어감

    RIO_BUF receive_buffer{};
    receive_buffer.BufferId = recv_buffer_id_;
    receive_buffer.Offset = static_cast<DWORD>(client->buffer_offset);
    receive_buffer.Length = static_cast<DWORD>(buffer_size_);

    DWORD flags = 0;
    client->is_receiving = true; // 완료가 즉시 dequeue되어도 상태가 먼저 보이게 한다.
    outstanding_requests_.fetch_add(1, std::memory_order_relaxed);
    if (!rio_fn_table_.RIOReceive(client->request_queue, &receive_buffer, 1, flags, client)) {
        client->is_receiving = false;
        outstanding_requests_.fetch_sub(1, std::memory_order_relaxed);
        return std::unexpected(GetLastErrorMessage());
    }

    return {};
}

std::expected<void, std::string> RIOEchoServer::PostSend(
    ClientContext* client, size_t data_size, size_t data_offset) {
    if (!client || client->request_queue == RIO_INVALID_RQ) {
        return std::unexpected("유효하지 않은 클라이언트");
    }
    if (client->is_sending) return std::unexpected("송신 요청이 이미 진행 중입니다");
    if (data_size == 0 || data_offset > buffer_size_ || data_size > buffer_size_ - data_offset) {
        return std::unexpected("송신할 데이터 범위가 클라이언트 버퍼를 벗어났습니다");
    }

    std::memcpy(send_buffer_.data() + client->buffer_offset + data_offset,
                recv_buffer_.data() + client->buffer_offset + data_offset, data_size);

    RIO_BUF send_buffer{};
    send_buffer.BufferId = send_buffer_id_;
    send_buffer.Offset = static_cast<DWORD>(client->buffer_offset + data_offset);
    send_buffer.Length = static_cast<DWORD>(data_size);

    DWORD flags = 0;
    client->is_sending = true; // 완료가 즉시 dequeue되어도 상태가 먼저 보이게 한다.
    outstanding_requests_.fetch_add(1, std::memory_order_relaxed);
    if (!rio_fn_table_.RIOSend(client->request_queue, &send_buffer, 1, flags, client)) {
        client->is_sending = false;
        outstanding_requests_.fetch_sub(1, std::memory_order_relaxed);
        return std::unexpected(GetLastErrorMessage());
    }

    return {};
}

완료가 수신인지 송신인지, 플래그로 구분하기

HandleCompletion()이 완료 결과를 받았을 때, 그 완료가 방금 끝난 수신인지 송신인지 알아야 다음 행동(에코 전송 또는 다음 수신 예약)을 정할 수 있습니다. 이 서버는 3장에서 썼던 RecvContext/SendContext(세션마다 수신·송신용 컨텍스트를 따로 두는 방식) 대신, is_receiving/is_sending 두 불(bool) 값의 조합으로 완료 종류를 추론합니다.

그림 7-1. 두 상태를 번갈아 오가는 것을 전제로, is_receiving/is_sending 조합만으로 완료 종류를 구분합니다

void RIOEchoServer::ProcessCompletions(std::stop_token stop_token) {
    constexpr DWORD max_results = 32;
    std::vector<RIORESULT> results(max_results);

    while (!stop_token.stop_requested() && (running_ || outstanding_requests_ != 0)) {
        ULONG num_results = 0;

        if (use_event_notification_) {
            const int notify_result = rio_fn_table_.RIONotify(completion_queue_);
            if (notify_result != ERROR_SUCCESS) {
                LogError(std::format("RIONotify 실패: {}", GetErrorMessage(notify_result)));
                std::this_thread::sleep_for(std::chrono::milliseconds(100));
                continue;
            }

            DWORD wait_result = WaitForSingleObject(completion_event_, 1000);
            if (wait_result == WAIT_OBJECT_0) {
                num_results = rio_fn_table_.RIODequeueCompletion(completion_queue_, results.data(), max_results);
            } else if (wait_result == WAIT_TIMEOUT) {
                continue;
            } else {
                LogError(std::format("완료 이벤트 대기 실패: {}", GetLastErrorMessage()));
                std::this_thread::sleep_for(std::chrono::milliseconds(100));
                continue;
            }
        } else {
            num_results = rio_fn_table_.RIODequeueCompletion(completion_queue_, results.data(), max_results);
            if (num_results == 0) {
                std::this_thread::sleep_for(std::chrono::milliseconds(1));
                continue;
            }
        }

        if (num_results == RIO_CORRUPT_CQ) {
            LogError("완료 큐 손상: 메모리 손상 또는 RIO API 사용 오류를 점검해야 합니다");
            break;
        }

        if (num_results > 0) HandleCompletion(results.data(), num_results);
    }
}

void RIOEchoServer::HandleCompletion(RIORESULT* results, ULONG result_count) {
    for (ULONG i = 0; i < result_count; ++i) {
        RIORESULT& result = results[i];
        outstanding_requests_.fetch_sub(1, std::memory_order_relaxed);
        ClientContext* client = reinterpret_cast<ClientContext*>(result.RequestContext);

        if (!client) {
            LogError("유효하지 않은 클라이언트 컨텍스트");
            continue;
        }
        if (result.Status != ERROR_SUCCESS) {
            LogError(std::format("RIO 완료 실패: {}", GetErrorMessage(static_cast<DWORD>(result.Status))));
            RemoveClient(client);
            continue;
        }
        if (!running_) {
            RemoveClient(client);
            continue;
        }
        if (result.BytesTransferred == 0) {
            RemoveClient(client);
            continue;
        }

        if (client->is_receiving && !client->is_sending) {
            client->is_receiving = false;
            LogInfo(std::format("데이터 수신: {} 바이트", result.BytesTransferred));

            client->send_offset = 0;
            client->send_remaining = result.BytesTransferred;
            auto send_result = PostSend(client, client->send_remaining, client->send_offset);
            if (!send_result) {
                LogError(std::format("송신 요청 실패: {}", send_result.error()));
                RemoveClient(client);
            }
        } else if (client->is_sending && !client->is_receiving) {
            client->is_sending = false;
            LogInfo(std::format("데이터 송신: {} 바이트", result.BytesTransferred));

            if (result.BytesTransferred > client->send_remaining) {
                LogError("송신 완료 바이트 수가 요청 범위를 벗어났습니다");
                RemoveClient(client);
                continue;
            }

            client->send_offset += result.BytesTransferred;
            client->send_remaining -= result.BytesTransferred;
            if (client->send_remaining != 0) {
                auto send_result = PostSend(client, client->send_remaining, client->send_offset);
                if (!send_result) {
                    LogError(std::format("나머지 송신 요청 실패: {}", send_result.error()));
                    RemoveClient(client);
                }
            } else {
                auto recv_result = PostReceive(client);
                if (!recv_result) {
                    LogError(std::format("수신 요청 실패: {}", recv_result.error()));
                    RemoveClient(client);
                }
            }
        } else {
            LogError("예상치 못한 클라이언트 상태");
            RemoveClient(client);
        }
    }
}

이 방식이 통하는 이유, 그리고 깨지는 조건

PostSend()는 이미 송신 중이면 std::unexpected를 반환하고, HandleCompletion()은 수신 완료 처리 후에만 PostSend()를, 송신 완료 처리 후에만 PostReceive()를 호출합니다. 즉 코드 스스로 "수신 → 송신 → 수신 → 송신 …" 순서를 강제하기 때문에, 두 상태가 동시에 참이 되는 일이 없다는 전제가 성립하고, 그 덕분에 bool 두 개만으로 완료 종류를 안전하게 구분할 수 있습니다. 하지만 이 서버를 확장해 수신과 송신을 동시에(전이중, full-duplex) 진행해야 하는 상황 — 예를 들어 클라이언트가 계속 이동 패킷을 보내는 동안 서버도 독립적으로 브로드캐스트 메시지를 밀어 넣어야 하는 실제 게임 서버 — 이 되면 이 전제가 깨지고, HandleCompletion()의 else 분기("예상치 못한 상태")로 빠져 연결이 끊깁니다. 실전에서 다중 동시 요청을 허용하려면 3장에서 배운 RecvContext/SendContext처럼 요청마다 고유한 RequestContext를 넘겨 완료 종류를 직접 구분하는 편이 훨씬 견고합니다.

송신은 한 번에 전부 끝난다고 가정하면 안 됩니다

TCP 송신 완료의 BytesTransferred는 요청한 길이보다 작을 수 있습니다. 따라서 예제는 send_offset과 send_remaining을 유지해 남은 바이트를 다시 PostSend()합니다. 완료 결과는 먼저 Status가 성공인지 확인한 뒤에만 바이트 수를 해석합니다.

7.4 오류 처리 및 예외 관리

APIFormatMessageA()

지금까지의 모든 장에서 오류가 나면 WSAGetLastError()가 돌려주는 숫자 코드만 출력했습니다. FormatMessageA()는 이 숫자를 Windows가 미리 등록해둔 사람이 읽을 수 있는 오류 문장으로 바꿔줍니다 — 예를 들어 코드 10054는 "원격 호스트에 의해 기존 연결이 강제로 끊겼습니다"처럼 번역됩니다.

매개변수의미
FORMAT_MESSAGE_ALLOCATE_BUFFER메시지를 담을 버퍼를 함수가 직접 할당하게 함(호출자는 LocalFree()로 해제).
FORMAT_MESSAGE_FROM_SYSTEMWindows 시스템이 미리 등록한 오류 메시지 테이블에서 찾음.
FORMAT_MESSAGE_IGNORE_INSERTS메시지에 %1 같은 삽입 인자가 있어도 무시하고 그대로 반환.
std::string RIOEchoServer::GetErrorMessage(DWORD error_code) const {
    if (error_code == 0) return "알 수 없는 오류";

    char* message_buffer = nullptr;
    DWORD message_size = FormatMessageA(
        FORMAT_MESSAGE_ALLOCATE_BUFFER | FORMAT_MESSAGE_FROM_SYSTEM | FORMAT_MESSAGE_IGNORE_INSERTS,
        nullptr, error_code, MAKELANGID(LANG_NEUTRAL, SUBLANG_DEFAULT),
        reinterpret_cast<LPSTR>(&message_buffer), 0, nullptr
    );

    if (message_buffer == nullptr) return std::format("오류 코드: {}", error_code);

    std::string error_message(message_buffer, message_size);
    LocalFree(message_buffer);   // FORMAT_MESSAGE_ALLOCATE_BUFFER로 받은 메모리는 직접 해제해야 함

    if (!error_message.empty() && error_message.back() == '\n') error_message.pop_back();
    if (!error_message.empty() && error_message.back() == '\r') error_message.pop_back();

    return std::format("오류 코드: {} - {}", error_code, error_message);
}

std::string RIOEchoServer::GetLastErrorMessage() const {
    return GetErrorMessage(WSAGetLastError());
}

void RIOEchoServer::LogInfo(std::string_view message) { std::print("[정보] {}\n", message); }
void RIOEchoServer::LogError(std::string_view message) { std::print("[오류] {}\n", message); }

줄바꿈을 제거하는 두 줄이 있는 이유

Windows 시스템 오류 메시지는 끝에 \r\n이 붙어 오는 경우가 많습니다. 이를 그대로 로그에 출력하면 한 줄짜리 오류 로그가 중간에 개행되어 보기 불편해집니다. pop_back() 두 번으로 끝의 개행 문자를 제거해, 로그 한 줄에 오류 메시지 하나가 깔끔하게 나오도록 합니다.

안전한 예외 처리 전략

이 책의 다른 코드는 대부분 std::expected로 예외 없이 오류를 처리하지만, main() 최상위에는 여전히 try/catch를 둡니다. 6장에서 봤듯 RIOCompletionQueue 같은 클래스의 생성자가 std::runtime_error를 던질 수 있고, 예상치 못한 표준 라이브러리 예외(예: std::bad_alloc)가 발생할 수도 있기 때문입니다. 

void RunEchoServerSafely(uint16_t port = 9000) {
    try {
        std::print("RIO 에코 서버 시작 중...\n");

        RIOEchoServer server(port);

        auto init_result = server.Initialize();
        if (!init_result) {
            std::print("서버 초기화 실패: {}\n", init_result.error());
            return;
        }

        auto start_result = server.Start();
        if (!start_result) {
            std::print("서버 시작 실패: {}\n", start_result.error());
            return;
        }

        std::print("서버가 포트 {}에서 실행 중입니다. 종료하려면 Enter 키를 누르세요...\n", port);
        std::cin.get();

        server.Stop();
        std::print("서버가 정상적으로 종료되었습니다.\n");

    } catch (const std::exception& e) {
        std::print("예외 발생: {}\n", e.what());
    } catch (...) {
        std::print("알 수 없는 예외 발생\n");
    }
}

int main() {
    RunEchoServerSafely(9000);
    return 0;
}

catch (...) 가 마지막에 있는 이유

catch (const std::exception&)는 표준 라이브러리가 던지는 대부분의 예외를 잡지만, 이론적으로는 std::exception을 상속하지 않는 임의의 타입도 예외로 던질 수 있습니다(예: 정수, 문자열 리터럴). catch (...)는 그런 "무엇인지 알 수 없는" 예외까지 모두 잡아, 서버 프로세스가 처리되지 않은 예외로 인해 아예 죽어버리는 최악의 상황을 막는 마지막 안전망입니다.

  1. 비동기 I/O 처리 — RIO의 요청/완료 모델을 활용한 효율적인 처리.

  6. 모던 C++23 활용 — 최신 C++ 기능으로 안전하고 가독성 높은 코드 작성.

7.6 요약

  1. 완결된 구조 — 선언과 구현을 나눠 RIO 서버를 처음부터 끝까지 완성하는 흐름을 한 번 더 확인했습니다.
