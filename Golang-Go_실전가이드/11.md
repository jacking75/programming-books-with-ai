**11.1 제네릭스의 탄생 배경 — "왜 Go에 제네릭스가 필요한가?"**

Go는 처음부터 단순함을 핵심 철학으로 삼아 설계된 언어입니다. 그러나 제네릭스가 없던 시절, Go 개발자들은 타입에 구애받지 않는 범용 코드를 작성하기 위해 두 가지 불편한 방법 중 하나를 선택해야 했습니다.

첫 번째는 `interface{}`(또는 Go 1.18 이후의 `any`) 를 사용하는 방법입니다. 이 방식은 컴파일 타임 타입 체크를 완전히 포기하는 대신 런타임 타입 단언(type assertion)이 필요하며, 실수가 있을 경우 패닉(panic)이 발생합니다. 두 번째는 코드 중복입니다. `int`를 위한 `SumInt`, `float64`를 위한 `SumFloat64` 처럼 기능은 동일하지만 타입만 다른 함수를 여러 개 작성하는 방법으로, 유지보수가 매우 힘들어집니다.

아래 코드는 제네릭스 이전의 전형적인 고통을 보여줍니다.

```go
// 제네릭스 이전 방식 1: interface{} 사용 (타입 안전성 없음)
func SumAny(values []any) any {
    var total float64
    for _, v := range values {
        switch val := v.(type) {
        case int:
            total += float64(val)
        case float64:
            total += val
        // 새 타입이 추가될 때마다 여기를 수정해야 함 ← 위험!
        default:
            panic("unsupported type")
        }
    }
    return total
}

// 제네릭스 이전 방식 2: 타입별 중복 함수 (DRY 위반)
func SumInt(values []int) int {
    total := 0
    for _, v := range values {
        total += v
    }
    return total
}

func SumFloat64(values []float64) float64 {
    total := 0.0
    for _, v := range values {
        total += v
    }
    return total
}
// float32, int64, int32... 도 필요하면 또 작성해야 한다
```

제네릭스를 사용하면 이 모든 문제가 단 하나의 함수로 해결됩니다.

```go
// 제네릭스 이후: 타입 안전 + 재사용 가능
func Sum[T int | float64](values []T) T {
    var total T
    for _, v := range values {
        total += v
    }
    return total
}
```

이처럼 제네릭스는 "타입 안전성을 유지하면서 코드를 재사용"하는 강력한 도구입니다.

**11.2 타입 파라미터 기초 문법**

**11.2.1 제네릭 함수 선언**

제네릭 함수는 함수 이름 뒤에 대괄호 `[T 제약]` 형식으로 타입 파라미터를 선언합니다. 타입 파라미터는 일반적으로 `T`, `K`, `V`, `E` 같은 짧은 대문자 식별자를 사용하는 것이 Go 커뮤니티의 관례입니다.

```go
package main

import "fmt"

// [T any] : T는 어떤 타입도 될 수 있다는 의미
func Map[T, U any](s []T, f func(T) U) []U {
    result := make([]U, len(s))
    for i, v := range s {
        result[i] = f(v)
    }
    return result
}

func Filter[T any](s []T, f func(T) bool) []T {
    var result []T
    for _, v := range s {
        if f(v) {
            result = append(result, v)
        }
    }
    return result
}

func Reduce[T, U any](s []T, init U, f func(U, T) U) U {
    acc := init
    for _, v := range s {
        acc = f(acc, v)
    }
    return acc
}

func main() {
    nums := []int{1, 2, 3, 4, 5}

    // Map: int → string
    strs := Map(nums, func(n int) string {
        return fmt.Sprintf("num_%d", n)
    })
    fmt.Println(strs) // [num_1 num_2 num_3 num_4 num_5]

    // Filter: 짝수만 추출
    evens := Filter(nums, func(n int) bool {
        return n%2 == 0
    })
    fmt.Println(evens) // [2 4]

    // Reduce: 합계 계산
    sum := Reduce(nums, 0, func(acc, n int) int {
        return acc + n
    })
    fmt.Println(sum) // 15
}
```

**11.2.2 타입 추론 (Type Inference)**

Go 컴파일러는 대부분의 경우 함수 인수로부터 타입 파라미터를 자동으로 추론합니다. 따라서 `Map[int, string](nums, ...)` 처럼 타입을 명시하지 않아도 됩니다. 단, 추론이 불가능한 경우에는 명시해야 합니다.

```go
// 타입 추론이 동작하는 경우
result := Map(nums, func(n int) string { return fmt.Sprint(n) })

// 타입 추론이 불가능한 경우 (인수가 없을 때) → 명시 필요
func Zero[T any]() T {
    var zero T
    return zero
}

z := Zero[int]()     // 명시 필요 (int로 추론할 단서가 없음)
s := Zero[string]()  // 마찬가지
```

**11.2.3 제네릭 타입 (Generic Types)**

함수뿐만 아니라 구조체(struct), 인터페이스, 타입 별칭 등도 제네릭으로 선언할 수 있습니다.

```go
// 제네릭 스택 구현
type Stack[T any] struct {
    items []T
}

func (s *Stack[T]) Push(item T) {
    s.items = append(s.items, item)
}

func (s *Stack[T]) Pop() (T, bool) {
    if len(s.items) == 0 {
        var zero T
        return zero, false
    }
    last := s.items[len(s.items)-1]
    s.items = s.items[:len(s.items)-1]
    return last, true
}

func (s *Stack[T]) Peek() (T, bool) {
    if len(s.items) == 0 {
        var zero T
        return zero, false
    }
    return s.items[len(s.items)-1], true
}

func (s *Stack[T]) Len() int {
    return len(s.items)
}

func main() {
    // int 스택
    intStack := &Stack[int]{}
    intStack.Push(10)
    intStack.Push(20)
    intStack.Push(30)
    
    v, ok := intStack.Pop()
    fmt.Println(v, ok) // 30 true

    // string 스택 — 완전히 다른 타입이지만 같은 코드
    strStack := &Stack[string]{}
    strStack.Push("hello")
    strStack.Push("world")
    
    top, _ := strStack.Peek()
    fmt.Println(top) // world
}
```

**11.3 제약 (Constraints) — 타입 파라미터의 경계를 정의하다**

제약(Constraint)은 타입 파라미터 `T`에 허용되는 타입의 집합을 정의하는 인터페이스입니다. 제네릭스의 진정한 힘은 이 제약을 얼마나 정교하게 설계하느냐에 달려 있습니다.

```
┌─────────────────────────────────────────────────────────────┐
│                       제약의 계층 구조                        │
│                                                             │
│   any (= interface{})                                       │
│   └─ 모든 타입을 허용, == 연산 불가                           │
│                                                             │
│   comparable                                               │
│   └─ == 와 != 연산 가능 (map key 로 사용 가능)                │
│      ├─ bool, int, float, complex, string                   │
│      ├─ pointer, channel, interface                         │
│      └─ comparable 필드만으로 구성된 struct                   │
│                                                             │
│   cmp.Ordered                                               │
│   └─ <, <=, >, >= 연산 가능 (comparable의 부분 집합)          │
│      ├─ 정수형: int, int8, int16, int32, int64              │
│      ├─ 부호없는 정수: uint, uint8, ..., uintptr            │
│      ├─ 부동소수점: float32, float64                         │
│      └─ string                                              │
│                                                             │
│   사용자 정의 제약 (Custom Constraint)                        │
│   └─ 메서드 집합, 타입 집합, 또는 그 조합                      │
└─────────────────────────────────────────────────────────────┘
```

**11.3.1 `any` 제약**

`any`는 `interface{}`의 별칭으로, 모든 타입을 허용합니다. 타입 파라미터에 어떤 연산도 요구하지 않을 때 사용합니다.

```go
// any 제약: 타입에 상관없이 슬라이스의 첫 번째 요소 반환
func First[T any](s []T) (T, bool) {
    if len(s) == 0 {
        var zero T
        return zero, false
    }
    return s[0], true
}

// Pointer, Map 등 어떤 타입도 가능
first, ok := First([][]byte{{1, 2}, {3, 4}})
```

**11.3.2 `comparable` 제약**

`comparable`은 `==`와 `!=` 연산자를 사용할 수 있는 타입의 집합입니다. `map`의 키 타입으로 사용할 때 필수적으로 요구됩니다.

```go
// comparable: map 기반의 제네릭 Set 구현
type Set[T comparable] struct {
    items map[T]struct{}
}

func NewSet[T comparable]() *Set[T] {
    return &Set[T]{items: make(map[T]struct{})}
}

func (s *Set[T]) Add(item T) {
    s.items[item] = struct{}{}
}

func (s *Set[T]) Contains(item T) bool {
    _, ok := s.items[item]
    return ok
}

func (s *Set[T]) Remove(item T) {
    delete(s.items, item)
}

func (s *Set[T]) Len() int {
    return len(s.items)
}

// 두 Set의 교집합 반환
func Intersect[T comparable](a, b *Set[T]) *Set[T] {
    result := NewSet[T]()
    for item := range a.items {
        if b.Contains(item) {
            result.Add(item)
        }
    }
    return result
}

func main() {
    s1 := NewSet[string]()
    s1.Add("go")
    s1.Add("rust")
    s1.Add("python")

    s2 := NewSet[string]()
    s2.Add("go")
    s2.Add("java")
    s2.Add("python")

    inter := Intersect(s1, s2)
    fmt.Println(inter.Contains("go"))     // true
    fmt.Println(inter.Contains("rust"))   // false
}
```

**11.3.3 `cmp.Ordered` 제약**

Go 1.21에서 표준 라이브러리 `cmp` 패키지와 함께 `cmp.Ordered`가 도입되었습니다. 이 제약은 `<`, `<=`, `>`, `>=` 비교 연산이 가능한 모든 타입을 나타냅니다. 이전에는 `golang.org/x/exp/constraints` 패키지의 `Ordered`를 사용했지만 이제는 표준 라이브러리에 포함됩니다.

```go
package main

import (
    "cmp"
    "fmt"
)

// cmp.Ordered: 비교 가능한 타입에 대한 최솟값/최댓값 함수
func Min[T cmp.Ordered](a, b T) T {
    if a < b {
        return a
    }
    return b
}

func Max[T cmp.Ordered](a, b T) T {
    if a > b {
        return a
    }
    return b
}

// 슬라이스에서 최솟값과 최댓값을 동시에 반환
func MinMax[T cmp.Ordered](s []T) (min, max T) {
    if len(s) == 0 {
        panic("empty slice")
    }
    min, max = s[0], s[0]
    for _, v := range s[1:] {
        if v < min {
            min = v
        }
        if v > max {
            max = v
        }
    }
    return
}

// 제네릭 이진 탐색
func BinarySearch[T cmp.Ordered](sorted []T, target T) int {
    lo, hi := 0, len(sorted)-1
    for lo <= hi {
        mid := (lo + hi) / 2
        switch cmp.Compare(sorted[mid], target) {
        case 0:
            return mid
        case -1:
            lo = mid + 1
        case 1:
            hi = mid - 1
        }
    }
    return -1
}

func main() {
    fmt.Println(Min(3, 7))         // 3
    fmt.Println(Min(3.14, 2.71))   // 2.71
    fmt.Println(Min("apple", "banana")) // apple

    ints := []int{3, 1, 4, 1, 5, 9, 2, 6}
    // 정렬 후 이진 탐색
    // (실제로는 slices.Sort 사용)
    sorted := []int{1, 1, 2, 3, 4, 5, 6, 9}
    idx := BinarySearch(sorted, 5)
    fmt.Println(idx) // 5
    
    min, max := MinMax([]float64{3.14, 1.41, 2.71, 1.73})
    fmt.Printf("min=%.2f, max=%.2f\n", min, max) // min=1.41, max=3.14
}
```

**11.3.4 타입 집합(Type Set)을 이용한 사용자 정의 제약**

인터페이스 안에 `~T` (언더라잉 타입이 T인 모든 타입) 또는 `T1 | T2` (합집합) 문법으로 정교한 제약을 정의할 수 있습니다.

```go
package main

import "fmt"

// 정수 타입만 허용하는 커스텀 제약
type Integer interface {
    ~int | ~int8 | ~int16 | ~int32 | ~int64 |
        ~uint | ~uint8 | ~uint16 | ~uint32 | ~uint64 | ~uintptr
}

// 부동소수점 타입만 허용하는 커스텀 제약
type Float interface {
    ~float32 | ~float64
}

// 숫자 타입 전체를 허용하는 커스텀 제약
type Number interface {
    Integer | Float
}

// ~ 의 의미: 기반(underlying) 타입이 int인 사용자 정의 타입도 포함
type Celsius float64
type Fahrenheit float64

// ~ 덕분에 Celsius, Fahrenheit도 Float 제약을 만족함
func Abs[T Float](v T) T {
    if v < 0 {
        return -v
    }
    return v
}

// 메서드 제약: Stringer 인터페이스를 구현하는 타입
type Stringer interface {
    String() string
}

// 메서드 제약 + 타입 파라미터 조합
func PrintAll[T Stringer](items []T) {
    for _, item := range items {
        fmt.Println(item.String())
    }
}

// 여러 제약의 조합: Number이면서 fmt.Stringer는 아니지만
// 산술 연산 가능한 숫자
func Sum[T Number](values []T) T {
    var total T
    for _, v := range values {
        total += v
    }
    return total
}

func main() {
    fmt.Println(Sum([]int{1, 2, 3, 4, 5}))            // 15
    fmt.Println(Sum([]float64{1.1, 2.2, 3.3}))        // 6.6000...
    fmt.Println(Sum([]uint{10, 20, 30}))               // 60

    temp := Celsius(-36.6)
    fmt.Println(Abs(temp)) // 36.6
}
```

`~` (틸드) 연산자는 Go 제네릭스에서 가장 강력한 기능 중 하나입니다. `~int`는 "기반 타입(underlying type)이 `int`인 모든 타입"을 의미하므로, `type MyInt int` 처럼 선언된 사용자 정의 타입도 제약을 만족합니다. 이 덕분에 도메인 타입을 명확하게 정의하면서도 제네릭 함수를 재사용할 수 있습니다.

**11.4 표준 라이브러리의 제네릭 패키지**

Go 1.21부터 표준 라이브러리에 제네릭을 활용한 패키지들이 공식적으로 추가되었습니다. 이들을 잘 활용하면 반복적인 보일러플레이트 코드를 대폭 줄일 수 있습니다.

```
┌────────────────────────────────────────────────────────────┐
│           Go 표준 라이브러리 제네릭 패키지 (1.21+)           │
│                                                            │
│  ┌──────────┬─────────────────────────────────────────┐   │
│  │ 패키지   │ 주요 기능                                │   │
│  ├──────────┼─────────────────────────────────────────┤   │
│  │ slices   │ Sort, Contains, Index, Reverse, Compact │   │
```

**11.4.1 `slices` 패키지**

```go
package main

import (
    "cmp"
    "fmt"
    "slices"
)

type Person struct {
    Name string
    Age  int
}

func main() {
    // --- slices 패키지 ---
    nums := []int{3, 1, 4, 1, 5, 9, 2, 6, 5}

    // 정렬
    slices.Sort(nums)
    fmt.Println(nums) // [1 1 2 3 4 5 5 6 9]

    // 이진 탐색 (정렬된 슬라이스에서)
    idx, found := slices.BinarySearch(nums, 5)
    fmt.Println(idx, found) // 5 true

    // 중복 제거 (정렬 후 사용)
    compact := slices.Compact(nums)
    fmt.Println(compact) // [1 2 3 4 5 6 9]

    // 포함 여부
    fmt.Println(slices.Contains(nums, 9)) // true

    // 커스텀 비교자로 구조체 정렬
    people := []Person{
        {"Alice", 30},
        {"Bob", 25},
        {"Charlie", 35},
    }

    // 나이 기준 오름차순 정렬
    slices.SortFunc(people, func(a, b Person) int {
        return cmp.Compare(a.Age, b.Age)
    })
    fmt.Println(people) // [{Bob 25} {Alice 30} {Charlie 35}]

    // 최댓값 (cmp.Ordered 제약 사용)
    maxVal := slices.Max([]float64{3.14, 2.71, 1.41})
    fmt.Println(maxVal) // 3.14

    // 역순 뒤집기
    slices.Reverse(compact)
    fmt.Println(compact) // [9 6 5 4 3 2 1]
}
```

**11.4.2 `maps` 패키지**

```go
package main

import (
    "fmt"
    "maps"
    "slices"
)

func main() {
    inventory := map[string]int{
        "apple":  50,
        "banana": 30,
        "cherry": 20,
    }

    // 키 목록 (순서 보장 없음 → 정렬)
    keys := slices.Sorted(maps.Keys(inventory))
    fmt.Println(keys) // [apple banana cherry]

    // 값 목록
    vals := slices.Collect(maps.Values(inventory))
    slices.Sort(vals)
    fmt.Println(vals) // [20 30 50]

    // 맵 복사 (깊은 복사 아님, 값이 포인터면 주의)
    copied := maps.Clone(inventory)
    copied["apple"] = 999
    fmt.Println(inventory["apple"], copied["apple"]) // 50 999 (독립적)

    // 맵 동등 비교
    m1 := map[string]int{"a": 1, "b": 2}
    m2 := map[string]int{"a": 1, "b": 2}
    fmt.Println(maps.Equal(m1, m2)) // true

    // 조건부 삭제
    maps.DeleteFunc(inventory, func(k string, v int) bool {
        return v < 30 // 재고가 30 미만인 항목 삭제
    })
    fmt.Println(inventory) // map[apple:50 banana:30]
```

**11.4.3 `cmp` 패키지**

```go
package main

import (
    "cmp"
    "fmt"

func main() {
    // cmp.Compare: -1, 0, 1 반환 (sort.Interface 스타일)
    fmt.Println(cmp.Compare(1, 2))   // -1
    fmt.Println(cmp.Compare(2, 2))   //  0
    fmt.Println(cmp.Compare(3, 2))   //  1
    fmt.Println(cmp.Compare("b", "a")) // 1

    // cmp.Or: 첫 번째 비어있지 않은(zero value가 아닌) 값을 반환
    // 설정값 우선순위 처리에 매우 유용!
    name := cmp.Or("", "", "default") // "default"
    fmt.Println(name)

    port := cmp.Or(0, 0, 8080) // 8080
    fmt.Println(port)

    // 실무 활용: 환경 변수 → 설정 파일 → 기본값 우선순위
    envPort := 0       // 환경 변수 없음
    cfgPort := 3000    // 설정 파일에 명시
    defaultPort := 8080

    finalPort := cmp.Or(envPort, cfgPort, defaultPort)
    fmt.Println(finalPort) // 3000
}
```

**11.5 실천 패턴 — 제네릭스를 현명하게 사용하는 방법**

이 절에서는 실무에서 자주 등장하는 제네릭 패턴들을 살펴봅니다.

**11.5.1 패턴 1: 옵션 타입 (Option / Maybe)**

Go는 `(T, error)` 또는 `(T, bool)` 패턴을 자주 사용하지만, 제네릭스를 이용하면 Rust의 `Option<T>`와 유사한 타입을 정의할 수 있습니다.

```go
package main

import (
    "errors"
    "fmt"
)

// Option[T]: 값이 있을 수도 없을 수도 있는 타입
type Option[T any] struct {
    value    T
    hasValue bool
}

func Some[T any](v T) Option[T] {
    return Option[T]{value: v, hasValue: true}
}

func None[T any]() Option[T] {
    return Option[T]{}
}

func (o Option[T]) Unwrap() T {
    if !o.hasValue {
        panic("option is None")
    }
    return o.value
}

func (o Option[T]) UnwrapOr(defaultVal T) T {
    if !o.hasValue {
        return defaultVal
    }
    return o.value
}

func (o Option[T]) IsNone() bool { return !o.hasValue }
func (o Option[T]) IsSome() bool { return o.hasValue }

// Map: Option 안의 값을 변환
func MapOption[T, U any](o Option[T], f func(T) U) Option[U] {
    if o.IsNone() {
        return None[U]()
    }
    return Some(f(o.value))
}

// 실제 사용 예시: DB에서 사용자 조회
type User struct {
    ID   int
    Name string
}

func findUserByID(id int) Option[User] {
    db := map[int]User{
        1: {1, "Alice"},
        2: {2, "Bob"},
    }
    if u, ok := db[id]; ok {
        return Some(u)
    }
    return None[User]()
}

func main() {
    user := findUserByID(1)
    if user.IsSome() {
        fmt.Println("Found:", user.Unwrap().Name) // Found: Alice
    }

    missing := findUserByID(999)
    name := MapOption(missing, func(u User) string { return u.Name })
    fmt.Println(name.UnwrapOr("Anonymous")) // Anonymous

    _ = errors.New("example") // import 사용
}
```

**11.5.2 패턴 2: 제네릭 Result 타입**

```go
package main

import "fmt"

// Result[T]: 성공 또는 실패를 나타내는 타입
type Result[T any] struct {
    value T
    err   error
}

func Ok[T any](v T) Result[T] {
    return Result[T]{value: v}
}

func Err[T any](err error) Result[T] {
    return Result[T]{err: err}
}

func (r Result[T]) IsOk() bool { return r.err == nil }

func (r Result[T]) Unwrap() (T, error) {
    return r.value, r.err
}

func (r Result[T]) UnwrapOr(defaultVal T) T {
    if r.err != nil {
        return defaultVal
    }
    return r.value
}

// AndThen: 성공 시 다음 연산을 체이닝
func AndThen[T, U any](r Result[T], f func(T) Result[U]) Result[U] {
    if !r.IsOk() {
        return Err[U](r.err)
    }
    return f(r.value)
}

func parseInt(s string) Result[int] {
    var n int
    _, err := fmt.Sscanf(s, "%d", &n)
    if err != nil {
        return Err[int](fmt.Errorf("invalid integer: %q", s))
    }
    return Ok(n)
}

func double(n int) Result[int] {
    if n < 0 {
        return Err[int](fmt.Errorf("negative number: %d", n))
    }
    return Ok(n * 2)
}

func main() {
    // 성공 체이닝
    result := AndThen(parseInt("21"), double)
    val, err := result.Unwrap()
    fmt.Println(val, err) // 42 <nil>

    // 실패 전파
    result2 := AndThen(parseInt("abc"), double)
    fmt.Println(result2.IsOk()) // false
    fmt.Println(result2.UnwrapOr(-1)) // -1
}
```

**11.5.3 패턴 3: 제네릭 캐시 (LRU Cache)**

```go
package main

import (
    "container/list"
    "fmt"
    "sync"
)

// LRUCache[K, V]: K는 comparable, V는 any
type LRUCache[K comparable, V any] struct {
    mu       sync.Mutex
    capacity int
    list     *list.List
    items    map[K]*list.Element
}

type cacheEntry[K comparable, V any] struct {
    key   K
    value V
}

func NewLRUCache[K comparable, V any](capacity int) *LRUCache[K, V] {
    return &LRUCache[K, V]{
        capacity: capacity,
        list:     list.New(),
        items:    make(map[K]*list.Element),
    }
}

func (c *LRUCache[K, V]) Get(key K) (V, bool) {
    c.mu.Lock()
    defer c.mu.Unlock()
    
    if elem, ok := c.items[key]; ok {
        c.list.MoveToFront(elem)
        return elem.Value.(*cacheEntry[K, V]).value, true
    }
    var zero V
    return zero, false
}

func (c *LRUCache[K, V]) Put(key K, value V) {
    c.mu.Lock()
    defer c.mu.Unlock()

    if elem, ok := c.items[key]; ok {
        c.list.MoveToFront(elem)
        elem.Value.(*cacheEntry[K, V]).value = value
        return
    }

    if c.list.Len() >= c.capacity {
        // 가장 오래된 항목 제거
        oldest := c.list.Back()
        if oldest != nil {
            c.list.Remove(oldest)
            delete(c.items, oldest.Value.(*cacheEntry[K, V]).key)
        }
    }

    entry := &cacheEntry[K, V]{key: key, value: value}
    elem := c.list.PushFront(entry)
    c.items[key] = elem
}

func main() {
    cache := NewLRUCache[string, int](3)
    cache.Put("a", 1)
    cache.Put("b", 2)
    cache.Put("c", 3)

    v, ok := cache.Get("a")
    fmt.Println(v, ok) // 1 true

    // "a"가 최근 사용됐으므로, "b"가 evict됨
    cache.Put("d", 4)
    
    _, ok = cache.Get("b")
    fmt.Println(ok) // false (evicted)
    
    v, ok = cache.Get("a")
    fmt.Println(v, ok) // 1 true (still exists)
}
```

**11.5.4 패턴 4: 함수형 파이프라인**

```go
package main

import "fmt"

// Pipeline: 함수를 체이닝하여 변환 파이프라인 구성
type Pipeline[T any] struct {
    value T
}

func NewPipeline[T any](v T) *Pipeline[T] {
    return &Pipeline[T]{value: v}
}

func Pipe[T, U any](p *Pipeline[T], f func(T) U) *Pipeline[U] {
    return &Pipeline[U]{value: f(p.value)}
}

func (p *Pipeline[T]) Result() T {
    return p.value
}

func main() {
    // int → string 파이프라인
    result := Pipe(
        Pipe(
            NewPipeline([]int{5, 3, 1, 4, 2}),
            func(s []int) []int {
                // 버블 정렬 (예시용)
                cp := make([]int, len(s))
                copy(cp, s)
                for i := 0; i < len(cp); i++ {
                    for j := 0; j < len(cp)-i-1; j++ {
                        if cp[j] > cp[j+1] {
                            cp[j], cp[j+1] = cp[j+1], cp[j]
                        }
                    }
                }
                return cp
            },
        ),
        func(s []int) string {
            return fmt.Sprintf("sorted: %v", s)
        },
    ).Result()
    
    fmt.Println(result) // sorted: [1 2 3 4 5]
}
```

**11.6 제네릭 메서드의 제약 — 알아야 할 한계**

Go 제네릭스에는 몇 가지 중요한 제약이 있습니다. 이를 알지 못하면 컴파일 오류와 씨름하게 됩니다.

```go
// ❌ 불가능: 메서드에는 새로운 타입 파라미터를 선언할 수 없음
type MyType struct{}

// 컴파일 오류! 메서드는 자신의 타입 파라미터를 새로 정의할 수 없다
// func (m MyType) Process[T any](v T) T { return v }

// ✅ 올바른 방법 1: 타입 자체를 제네릭으로 만들기
type Processor[T any] struct{}

func (p Processor[T]) Process(v T) T { return v }

// ✅ 올바른 방법 2: 메서드 대신 최상위 제네릭 함수 사용
func Process[T any](v T) T { return v }
```

```go
// ❌ 불가능: 제네릭 타입의 타입 스위치
func printType[T any](v T) {
    // 컴파일 오류: cannot use type switch on type parameter
    // switch v.(type) {
    // case int: ...
    // }
    
    // ✅ 올바른 방법: any로 변환 후 스위치
    switch any(v).(type) {
    case int:
        fmt.Println("int:", any(v).(int))
    case string:
        fmt.Println("string:", any(v).(string))
    default:
        fmt.Printf("other: %v\n", v)
    }
}
```

```
┌────────────────────────────────────────────────────────────────┐
│                  Go 제네릭스 주요 제한사항                       │
│                                                                │
│  1. 메서드에 새로운 타입 파라미터 선언 불가                       │
│     → 타입 자체를 제네릭으로 만들거나 최상위 함수 사용             │
│                                                                │
│  2. 제네릭 타입에 대한 직접 타입 스위치 불가                      │
│     → any(v).(type) 으로 우회                                  │
│                                                                │
│  3. 타입 파라미터 T의 값으로 make(), new() 불가                  │
│     → 제약에 factory 함수 포함하거나 reflect 사용               │
│                                                                │
│  4. 제네릭 타입의 메서드는 타입 인수 부분 집합 사용 불가           │
│     → 전체 타입 파라미터 명시 필요                               │
└────────────────────────────────────────────────────────────────┘
```

**11.7 언제 써야 하고, 언제 쓰지 말아야 하는가?**

제네릭스는 강력하지만 모든 상황에 적합하지는 않습니다. Go 팀의 공식 가이드라인과 커뮤니티 경험을 종합하면 다음과 같은 판단 기준을 세울 수 있습니다.

```
┌─────────────────────────────────────────────────────────────────┐
│              제네릭스 사용 결정 플로우차트                         │
│                                                                 │
│   코드에서 타입만 다르고 로직이 같은 반복이 있는가?                │
│              │                                                  │
│       ┌──────┴──────┐                                           │
│       │ Yes         │ No                                        │
│       ▼             ▼                                           │
│  인터페이스로      제네릭 불필요                                  │
│  해결 가능한가?    → 그냥 구체 타입 사용                          │
│       │                                                         │
│  ┌────┴────┐                                                    │
│  │ Yes     │ No                                                 │
│  ▼         ▼                                                    │
│  인터페이스  타입 파라미터가                                       │
│  사용 권장   필요한 연산인가?                                     │
│             (비교, 산술 등)                                      │
│                  │                                              │
│             ┌────┴────┐                                         │
│             │ Yes     │ No                                      │
│             ▼         ▼                                         │
│          제네릭스     any + 타입 단언                             │
│          적합!        또는 reflect                               │
└─────────────────────────────────────────────────────────────────┘
```

**제네릭스를 써야 할 때:**

- **자료구조 구현**: Stack, Queue, LinkedList, Set, Map, Tree 등 타입에 무관한 컨테이너를 만들 때, 제네릭스는 타입 안전성을 보장하면서 중복 코드를 없앱니다.
- **함수형 헬퍼**: `Map`, `Filter`, `Reduce`, `Contains` 등 슬라이스/컬렉션 조작 함수를 작성할 때 제네릭스는 매우 자연스럽게 맞아떨어집니다.
- **알고리즘**: 정렬, 탐색, 비교 등 타입에 독립적인 알고리즘을 구현할 때 `cmp.Ordered`와 같은 제약과 함께 사용하면 강력합니다.
- **동기화 래퍼**: 여러 타입의 값을 안전하게 동기화하는 래퍼(`sync.Map`의 타입 안전 버전 등)를 만들 때 효과적입니다.

**제네릭스를 피해야 할 때:**

- **단순 인터페이스로 충분할 때**: `io.Reader`, `fmt.Stringer`처럼 메서드 기반 추상화로 해결되는 경우 제네릭스를 도입하면 오히려 코드가 복잡해집니다.
- **단 한 번만 쓰이는 함수**: 특정 타입에만 쓰이는 함수를 "언젠가 재사용할 것 같다"는 이유로 제네릭으로 만드는 것은 과도한 추상화(over-engineering)입니다.
- **성능이 극도로 중요한 핫패스**: Go의 제네릭스는 GCShape stenciling 방식으로 구현되어 경우에 따라 인터페이스보다 빠르지만, 동일 GCShape를 공유하는 경우 런타임 딕셔너리를 사용해 약간의 오버헤드가 생길 수 있습니다. 성능이 극도로 중요하다면 벤치마크 후 결정해야 합니다.
- **팀의 제네릭스 이해도가 낮을 때**: 제네릭스 코드는 읽기 어려울 수 있습니다. 팀 전체가 이해할 수 있는 범위 내에서 사용해야 합니다.

**11.8 제네릭스와 인터페이스의 차이 — 핵심 비교**

```
┌──────────────────────────────────────────────────────────────────┐
│              인터페이스 vs 제네릭스 선택 기준                      │
│                                                                  │
│  상황: 여러 타입이 같은 메서드를 구현할 때                          │
│                                                                  │
│  인터페이스 (동적 디스패치)          제네릭스 (정적 단형화)          │
│  ─────────────────────────         ────────────────────────      │
│  + 런타임에 타입 결정               + 컴파일 타임에 타입 확정       │
│  + 새 타입 추가 시 코드 변경 없음   + 타입 안전성 완벽 보장         │
│  + 더 단순하고 관용적               + 불필요한 박싱(boxing) 없음    │
│  - 타입 단언 시 런타임 패닉 가능    - 코드 가독성 저하 가능          │
│  - 인터페이스 값 박싱 오버헤드      - 컴파일 시간 증가              │
│                                                                  │
│  ✅ 인터페이스가 적합한 경우:        ✅ 제네릭스가 적합한 경우:      │
│  - 메서드 행동을 추상화할 때        - 타입에 무관한 알고리즘         │
│  - 플러그인/전략 패턴              - 컨테이너/자료구조              │
│  - 런타임 다형성이 필요할 때        - 타입 안전 래퍼                │
└──────────────────────────────────────────────────────────────────┘
```

구체적인 비교 코드를 살펴보겠습니다.

```go
package main

import "fmt"

// 방법 1: 인터페이스 (행동 기반 추상화)
type Sizer interface {
    Size() int
}

func TotalSize(items []Sizer) int {
    total := 0
    for _, item := range items {
        total += item.Size()
    }
    return total
}

// 방법 2: 제네릭스 (타입 안전 컬렉션)
func Contains[T comparable](s []T, target T) bool {
    for _, v := range s {
        if v == target {
            return true
        }
    }
    return false
}

// ❌ 인터페이스로는 이게 불가능하다:
// comparable한 타입이라는 "능력"은 메서드로 표현할 수 없음
```

```go
package main

import (
    "cmp"
    "fmt"
)

// 여러 인터페이스를 조합한 복합 제약
type Sortable[T any] interface {
    Len() int
    Less(i, j int) bool
    Swap(i, j int)
}

// 비교 가능하면서 출력 가능한 타입
type ComparableStringer interface {
    comparable
    fmt.Stringer
}

// 숫자이면서 인코딩 가능한 타입 (실무적 복합 제약 예시)
type OrderedKey interface {
    cmp.Ordered
}

// 제네릭 우선순위 큐 (최소 힙)
type MinHeap[T cmp.Ordered] struct {
    data []T
}

func (h *MinHeap[T]) Push(v T) {
    h.data = append(h.data, v)
    h.siftUp(len(h.data) - 1)
}

func (h *MinHeap[T]) Pop() (T, bool) {
    if len(h.data) == 0 {
        var zero T
        return zero, false
    }
    min := h.data[0]
    last := len(h.data) - 1
    h.data[0] = h.data[last]
    h.data = h.data[:last]
    if len(h.data) > 0 {
        h.siftDown(0)
    }
    return min, true
}

func (h *MinHeap[T]) siftUp(i int) {
    for i > 0 {
        parent := (i - 1) / 2
        if h.data[i] < h.data[parent] {
            h.data[i], h.data[parent] = h.data[parent], h.data[i]
            i = parent
        } else {
            break
        }
    }
}

func (h *MinHeap[T]) siftDown(i int) {
    n := len(h.data)
    for {
        smallest := i
        left, right := 2*i+1, 2*i+2
        if left < n && h.data[left] < h.data[smallest] {
            smallest = left
        }
        if right < n && h.data[right] < h.data[smallest] {
            smallest = right
        }
        if smallest == i {
            break
        }
        h.data[i], h.data[smallest] = h.data[smallest], h.data[i]
        i = smallest
    }
}

func main() {
    heap := &MinHeap[int]{}
    for _, v := range []int{5, 3, 8, 1, 9, 2} {
        heap.Push(v)
    }

    for {
        v, ok := heap.Pop()
        if !ok {
            break
        }
        fmt.Printf("%d ", v)
    }
    // 출력: 1 2 3 5 8 9 (정렬된 순서)
    fmt.Println()

    // string 힙도 동일한 코드로 동작!
    strHeap := &MinHeap[string]{}
    for _, s := range []string{"banana", "apple", "cherry"} {
        strHeap.Push(s)
    }
    for {
        v, ok := strHeap.Pop()
        if !ok {
            break
        }
        fmt.Printf("%s ", v)
    }
    // 출력: apple banana cherry
    fmt.Println()
}
```

**11.10 제네릭스 성능 이해 — GCShape Stenciling**

Go 제네릭스의 내부 구현 방식을 이해하면 성능 특성을 예측할 수 있습니다.

```
┌─────────────────────────────────────────────────────────────────┐
│              Go 제네릭스 구현: GCShape Stenciling               │
│                                                                 │
│  C++처럼 타입별로 코드를 완전히 복제하지 않고,                    │
│  "GC 형태(GCShape)"가 같은 타입끼리는 코드를 공유한다.            │
│                                                                 │
│  GCShape가 같은 타입 그룹:                                       │
│  ┌─────────────────────────────────────────────────────┐       │
│  │ 포인터 타입 (*T)  → 모두 하나의 코드 공유              │       │
│  │ int, int8, int16  → 각각 별도 (크기가 다름)           │       │
│  │ struct{ x int }   → 같은 GCShape면 공유              │       │
│  └─────────────────────────────────────────────────────┘       │
│                                                                 │
│  포인터 타입은 하나의 코드를 공유하고 런타임 딕셔너리로 디스패치    │
│  → 이 경우 인터페이스와 유사한 오버헤드 발생 가능                  │
│                                                                 │
│  int, float64 같은 기본 스칼라 타입은 코드 특화(specialization)  │
│  → 박싱 없이 직접 연산 → 인터페이스보다 빠를 수 있음              │
└─────────────────────────────────────────────────────────────────┘
```

```go
// gendb/gendb.go
package main

import (
	"cmp"
	"errors"
	"fmt"
	"slices"
	"sync"
	"sync/atomic"
	"time"
)

// ──────────────────────────────────────────────
// 핵심 에러 정의
// ──────────────────────────────────────────────

var (
	ErrNotFound   = errors.New("record not found")
	ErrDuplicate  = errors.New("duplicate key")
	ErrEmptyTable = errors.New("collection is empty")
)

// ──────────────────────────────────────────────
// Record: 모든 데이터 레코드의 기반 인터페이스
// ──────────────────────────────────────────────

type Record interface {
	GetID() int64
}

// ──────────────────────────────────────────────
// Collection[T]: 제네릭 컬렉션 (인메모리 테이블)
// ──────────────────────────────────────────────

type Collection[T Record] struct {
	mu       sync.RWMutex
	items    map[int64]T
	nextID   atomic.Int64
	watchers []chan Event[T]
	wmu      sync.Mutex
}

// Event: 변경 이벤트 (Observable 패턴)
type EventType string

const (
	EventInsert EventType = "INSERT"
	EventUpdate EventType = "UPDATE"
	EventDelete EventType = "DELETE"
)

type Event[T any] struct {
	Type      EventType
	Record    T
	Timestamp time.Time
}

func NewCollection[T Record]() *Collection[T] {
	c := &Collection[T]{
		items: make(map[int64]T),
	}
	c.nextID.Store(1)
	return c
}

// Insert: 새 레코드 삽입, ID를 자동 할당하거나 기존 ID 사용
func (c *Collection[T]) Insert(item T) (int64, error) {
	c.mu.Lock()
	defer c.mu.Unlock()

	id := item.GetID()
	if id == 0 {
		id = c.nextID.Add(1) - 1
	}
	if _, exists := c.items[id]; exists {
		return 0, fmt.Errorf("%w: id=%d", ErrDuplicate, id)
	}
	c.items[id] = item
	c.notify(Event[T]{Type: EventInsert, Record: item, Timestamp: time.Now()})
	return id, nil
}

// FindByID: ID로 단건 조회
func (c *Collection[T]) FindByID(id int64) (T, error) {
	c.mu.RLock()
	defer c.mu.RUnlock()

	if item, ok := c.items[id]; ok {
		return item, nil
	}
	var zero T
	return zero, fmt.Errorf("%w: id=%d", ErrNotFound, id)
}

// Update: 레코드 업데이트
func (c *Collection[T]) Update(item T) error {
	c.mu.Lock()
	defer c.mu.Unlock()

	id := item.GetID()
	if _, exists := c.items[id]; !exists {
		return fmt.Errorf("%w: id=%d", ErrNotFound, id)
	}
	c.items[id] = item
	c.notify(Event[T]{Type: EventUpdate, Record: item, Timestamp: time.Now()})
	return nil
}

// Delete: 레코드 삭제
func (c *Collection[T]) Delete(id int64) error {
	c.mu.Lock()
	defer c.mu.Unlock()

	item, exists := c.items[id]
	if !exists {
		return fmt.Errorf("%w: id=%d", ErrNotFound, id)
	}
	delete(c.items, id)
	c.notify(Event[T]{Type: EventDelete, Record: item, Timestamp: time.Now()})
	return nil
}

// FindAll: 전체 레코드 반환
func (c *Collection[T]) FindAll() []T {
	c.mu.RLock()
	defer c.mu.RUnlock()

	result := make([]T, 0, len(c.items))
	for _, item := range c.items {
		result = append(result, item)
	}
	return result
}

// Count: 전체 레코드 수
func (c *Collection[T]) Count() int {
	c.mu.RLock()
	defer c.mu.RUnlock()
	return len(c.items)
}

// Watch: 변경 이벤트 구독
func (c *Collection[T]) Watch(bufSize int) <-chan Event[T] {
	ch := make(chan Event[T], bufSize)
	c.wmu.Lock()
	c.watchers = append(c.watchers, ch)
	c.wmu.Unlock()
	return ch
}

func (c *Collection[T]) notify(e Event[T]) {
	c.wmu.Lock()
	defer c.wmu.Unlock()
	for _, ch := range c.watchers {
		select {
		case ch <- e:
		default: // 버퍼가 꽉 찬 경우 드롭
		}
	}
}

// ──────────────────────────────────────────────
// Query[T]: 제네릭 쿼리 빌더 (플루언트 API)
// ──────────────────────────────────────────────

type Query[T Record] struct {
	collection *Collection[T]
	filters    []func(T) bool
	orderBy    func(a, b T) int
	limitVal   int
	offsetVal  int
}

func (c *Collection[T]) Query() *Query[T] {
	return &Query[T]{collection: c, limitVal: -1}
}

// Where: 필터 조건 추가 (체이닝 가능)
func (q *Query[T]) Where(predicate func(T) bool) *Query[T] {
	q.filters = append(q.filters, predicate)
	return q
}

// OrderBy: 정렬 함수 설정 (cmp.Compare 스타일)
func (q *Query[T]) OrderBy(less func(a, b T) int) *Query[T] {
	q.orderBy = less
	return q
}

// Limit: 결과 최대 개수 제한
func (q *Query[T]) Limit(n int) *Query[T] {
	q.limitVal = n
	return q
}

// Offset: 결과 시작 위치
func (q *Query[T]) Offset(n int) *Query[T] {
	q.offsetVal = n
	return q
}

// Execute: 쿼리 실행
func (q *Query[T]) Execute() []T {
	all := q.collection.FindAll()

	// 필터 적용
	filtered := Filter(all, func(item T) bool {
		for _, f := range q.filters {
			if !f(item) {
				return false
			}
		}
		return true
	})

	// 정렬 적용
	if q.orderBy != nil {
		slices.SortFunc(filtered, q.orderBy)
	}

	// Offset 적용
	if q.offsetVal > 0 {
		if q.offsetVal >= len(filtered) {
			return nil
		}
		filtered = filtered[q.offsetVal:]
	}

	// Limit 적용
	if q.limitVal >= 0 && q.limitVal < len(filtered) {
		filtered = filtered[:q.limitVal]
	}

	return filtered
}

// First: 첫 번째 결과만 반환
func (q *Query[T]) First() (T, error) {
	results := q.Limit(1).Execute()
	if len(results) == 0 {
		var zero T
		return zero, ErrNotFound
	}
	return results[0], nil
}

// ──────────────────────────────────────────────
// 제네릭 헬퍼 함수
// ──────────────────────────────────────────────

func Filter[T any](s []T, f func(T) bool) []T {
	result := make([]T, 0)
	for _, v := range s {
		if f(v) {
			result = append(result, v)
		}
	}
	return result
}

func Map[T, U any](s []T, f func(T) U) []U {
	result := make([]U, len(s))
	for i, v := range s {
		result[i] = f(v)
	}
	return result
}

func Reduce[T, U any](s []T, init U, f func(U, T) U) U {
	acc := init
	for _, v := range s {
		acc = f(acc, v)
	}
	return acc
}

// ──────────────────────────────────────────────
// Aggregator[T]: 집계 함수 모음
// ──────────────────────────────────────────────

type Aggregator[T any] struct {
	data []T
}

func NewAggregator[T any](data []T) *Aggregator[T] {
	return &Aggregator[T]{data: data}
}

// Sum: 수치 필드의 합계
func Sum[T any, N interface{ ~int | ~float64 }](
	data []T, selector func(T) N,
) N {
	return Reduce(data, N(0), func(acc N, item T) N {
		return acc + selector(item)
	})
}

// Avg: 수치 필드의 평균
func Avg[T any, N interface{ ~int | ~float64 }](
	data []T, selector func(T) N,
) float64 {
	if len(data) == 0 {
		return 0
	}
	total := Sum(data, selector)
	return float64(total) / float64(len(data))
}

// MinBy: 특정 필드 기준 최솟값을 가진 레코드 반환
func MinBy[T any, K cmp.Ordered](data []T, key func(T) K) (T, error) {
	if len(data) == 0 {
		var zero T
		return zero, ErrEmptyTable
	}
	return slices.MinFunc(data, func(a, b T) int {
		return cmp.Compare(key(a), key(b))
	}), nil
}

// MaxBy: 특정 필드 기준 최댓값을 가진 레코드 반환
func MaxBy[T any, K cmp.Ordered](data []T, key func(T) K) (T, error) {
	if len(data) == 0 {
		var zero T
		return zero, ErrEmptyTable
	}
	return slices.MaxFunc(data, func(a, b T) int {
		return cmp.Compare(key(a), key(b))
	}), nil
}

// GroupBy: 키 함수 기준 그룹화
func GroupBy[T any, K comparable](data []T, key func(T) K) map[K][]T {
	result := make(map[K][]T)
	for _, item := range data {
		k := key(item)
		result[k] = append(result[k], item)
	}
	return result
}

// ──────────────────────────────────────────────
// Index[K, V]: 제네릭 해시 인덱스
// ──────────────────────────────────────────────

type Index[K comparable, V Record] struct {
	mu      sync.RWMutex
	keyFunc func(V) K
	data    map[K][]int64 // key → record IDs
}

func NewIndex[K comparable, V Record](keyFunc func(V) K) *Index[K, V] {
	return &Index[K, V]{
		keyFunc: keyFunc,
		data:    make(map[K][]int64),
	}
}

func (idx *Index[K, V]) Add(item V) {
	idx.mu.Lock()
	defer idx.mu.Unlock()
	k := idx.keyFunc(item)
	idx.data[k] = append(idx.data[k], item.GetID())
}

func (idx *Index[K, V]) Remove(item V) {
	idx.mu.Lock()
	defer idx.mu.Unlock()
	k := idx.keyFunc(item)
	ids := idx.data[k]
	targetID := item.GetID()
	newIDs := Filter(ids, func(id int64) bool { return id != targetID })
	if len(newIDs) == 0 {
		delete(idx.data, k)
	} else {
		idx.data[k] = newIDs
	}
}

func (idx *Index[K, V]) Lookup(key K) []int64 {
	idx.mu.RLock()
	defer idx.mu.RUnlock()
	ids := idx.data[key]
	result := make([]int64, len(ids))
	copy(result, ids)
	return result
}

// ──────────────────────────────────────────────
// 도메인 모델 정의
// ──────────────────────────────────────────────

type Department string

const (
	DeptEngineering Department = "Engineering"
	DeptMarketing   Department = "Marketing"
	DeptSales       Department = "Sales"
	DeptHR          Department = "HR"
)

type Employee struct {
	ID         int64
	Name       string
	Department Department
	Salary     float64
	Level      int
	JoinedAt   time.Time
}

func (e Employee) GetID() int64 { return e.ID }

type Product struct {
	ID       int64
	Name     string
	Category string
	Price    float64
	Stock    int
}

func (p Product) GetID() int64 { return p.ID }

// ──────────────────────────────────────────────
// 메인: 통합 시연
// ──────────────────────────────────────────────

func main() {
	fmt.Println("╔══════════════════════════════════════════════════╗")
	fmt.Println("║   GenDB - Generic In-Memory Database Demo        ║")
	fmt.Println("╚══════════════════════════════════════════════════╝")
	fmt.Println()

	// ── 1. Employee 컬렉션 구성 ──────────────────────
	fmt.Println("▶ [1] Employee 컬렉션 구성")
	fmt.Println("─────────────────────────────────────────────────")

	employees := NewCollection[Employee]()

	// 변경 이벤트 구독 (비동기)
	events := employees.Watch(20)
	go func() {
		for e := range events {
			fmt.Printf("  📢 EVENT [%s] → %s (ID:%d)\n",
				e.Type, e.Record.Name, e.Record.ID)
		}
	}()

	// 직원 데이터 삽입
	sampleEmployees := []Employee{
		{ID: 1, Name: "Alice", Department: DeptEngineering, Salary: 95000, Level: 4, JoinedAt: mustParseDate("2020-03-15")},
		{ID: 2, Name: "Bob", Department: DeptEngineering, Salary: 85000, Level: 3, JoinedAt: mustParseDate("2021-06-01")},
		{ID: 3, Name: "Charlie", Department: DeptMarketing, Salary: 75000, Level: 3, JoinedAt: mustParseDate("2019-11-20")},
		{ID: 4, Name: "Diana", Department: DeptSales, Salary: 70000, Level: 2, JoinedAt: mustParseDate("2022-01-10")},
		{ID: 5, Name: "Eve", Department: DeptEngineering, Salary: 110000, Level: 5, JoinedAt: mustParseDate("2018-08-05")},
		{ID: 6, Name: "Frank", Department: DeptHR, Salary: 65000, Level: 2, JoinedAt: mustParseDate("2023-03-01")},
		{ID: 7, Name: "Grace", Department: DeptMarketing, Salary: 80000, Level: 4, JoinedAt: mustParseDate("2020-07-14")},
		{ID: 8, Name: "Hank", Department: DeptSales, Salary: 72000, Level: 3, JoinedAt: mustParseDate("2021-09-30")},
	}

	for _, emp := range sampleEmployees {
		id, err := employees.Insert(emp)
		if err != nil {
			fmt.Printf("  ❌ 삽입 실패: %v\n", err)
			continue
		}
		_ = id
	}

	time.Sleep(50 * time.Millisecond) // 이벤트 출력 대기
	fmt.Printf("\n  총 직원 수: %d명\n\n", employees.Count())

	// ── 2. 제네릭 인덱스를 활용한 조회 ────────────────
	fmt.Println("▶ [2] 부서 인덱스 구축 및 조회")
	fmt.Println("─────────────────────────────────────────────────")

	deptIndex := NewIndex[Department, Employee](func(e Employee) Department {
		return e.Department
	})
	for _, emp := range employees.FindAll() {
		deptIndex.Add(emp)
	}

	engineeringIDs := deptIndex.Lookup(DeptEngineering)
	fmt.Printf("  Engineering 부서 직원 IDs: %v\n", engineeringIDs)

	fmt.Println()

	// ── 3. 플루언트 쿼리 API ──────────────────────────
	fmt.Println("▶ [3] 플루언트 쿼리 API 시연")
	fmt.Println("─────────────────────────────────────────────────")

	// 쿼리 1: Engineering 부서, 급여 90000 이상, 급여 내림차순
	seniorEngineers := employees.Query().
		Where(func(e Employee) bool { return e.Department == DeptEngineering }).
		Where(func(e Employee) bool { return e.Salary >= 90000 }).
		OrderBy(func(a, b Employee) int { return cmp.Compare(b.Salary, a.Salary) }). // 내림차순
		Execute()

	fmt.Println("  🔍 시니어 엔지니어 (급여 ≥ 90,000, 내림차순):")
	for _, e := range seniorEngineers {
		fmt.Printf("     %-10s | %s | $%.0f | Level %d\n",
			e.Name, e.Department, e.Salary, e.Level)
	}

	// 쿼리 2: 2021년 이후 입사자, 이름 오름차순
	fmt.Println()
	recentHires := employees.Query().
		Where(func(e Employee) bool { return e.JoinedAt.Year() >= 2021 }).
		OrderBy(func(a, b Employee) int { return cmp.Compare(a.Name, b.Name) }).
		Execute()

	fmt.Println("  🔍 2021년 이후 입사자 (이름 오름차순):")
	for _, e := range recentHires {
		fmt.Printf("     %-10s | %s | joined: %s\n",
			e.Name, e.Department, e.JoinedAt.Format("2006-01-02"))
	}

	// 쿼리 3: 페이지네이션 (2번째 페이지, 3명씩)
	fmt.Println()
	page2 := employees.Query().
		OrderBy(func(a, b Employee) int { return cmp.Compare(a.ID, b.ID) }).
		Limit(3).
		Offset(3).
		Execute()

	fmt.Println("  🔍 페이지 2 (ID 기준, 3명씩):")
	for _, e := range page2 {
		fmt.Printf("     ID:%-2d %-10s\n", e.ID, e.Name)
	}

	fmt.Println()

	// ── 4. 제네릭 집계 함수 ───────────────────────────
	fmt.Println("▶ [4] 제네릭 집계 함수 시연")
	fmt.Println("─────────────────────────────────────────────────")

	allEmployees := employees.FindAll()

	// 전체 급여 합계
	totalSalary := Sum(allEmployees, func(e Employee) float64 { return e.Salary })
	fmt.Printf("  💰 전체 급여 합계   : $%.0f\n", totalSalary)

	// 평균 급여
	avgSalary := Avg(allEmployees, func(e Employee) float64 { return e.Salary })
	fmt.Printf("  💰 평균 급여       : $%.2f\n", avgSalary)

	// 최고 급여 직원
	highest, _ := MaxBy(allEmployees, func(e Employee) float64 { return e.Salary })
	fmt.Printf("  🏆 최고 급여 직원  : %s ($%.0f)\n", highest.Name, highest.Salary)

	// 최저 급여 직원
	lowest, _ := MinBy(allEmployees, func(e Employee) float64 { return e.Salary })
	fmt.Printf("  📉 최저 급여 직원  : %s ($%.0f)\n", lowest.Name, lowest.Salary)

	fmt.Println()

	// ── 5. GroupBy를 활용한 부서별 통계 ───────────────
	fmt.Println("▶ [5] GroupBy: 부서별 통계")
	fmt.Println("─────────────────────────────────────────────────")

	byDept := GroupBy(allEmployees, func(e Employee) Department {
		return e.Department
	})

	// 부서 이름 정렬 (재현 가능한 출력)
	deptNames := make([]Department, 0, len(byDept))
	for dept := range byDept {
		deptNames = append(deptNames, dept)
	}
	slices.SortFunc(deptNames, func(a, b Department) int {
		return cmp.Compare(string(a), string(b))
	})

	fmt.Printf("  %-15s | %5s | %10s | %10s\n", "부서", "인원", "평균급여", "총급여")
	fmt.Println("  " + "─────────────────────────────────────────────")
	for _, dept := range deptNames {
		members := byDept[dept]
		avg := Avg(members, func(e Employee) float64 { return e.Salary })
		total := Sum(members, func(e Employee) float64 { return e.Salary })
		fmt.Printf("  %-15s | %5d | $%9.2f | $%9.0f\n",
			dept, len(members), avg, total)
	}

	fmt.Println()

	// ── 6. Map/Filter/Reduce 파이프라인 ───────────────
	fmt.Println("▶ [6] Map / Filter / Reduce 파이프라인")
	fmt.Println("─────────────────────────────────────────────────")

	// 엔지니어링 팀의 이름 목록 추출 → 정렬
	engNames := Map(
		Filter(allEmployees, func(e Employee) bool {
			return e.Department == DeptEngineering
		}),
		func(e Employee) string { return e.Name },
	)
	slices.Sort(engNames)
	fmt.Printf("  Engineering 팀: %v\n", engNames)

	// 전체 레벨 합계
	totalLevel := Reduce(allEmployees, 0, func(acc int, e Employee) int {
		return acc + e.Level
	})
	fmt.Printf("  전체 레벨 합계: %d (평균: %.1f)\n",
		totalLevel, float64(totalLevel)/float64(len(allEmployees)))

	fmt.Println()

	// ── 7. Product 컬렉션 (다른 타입으로 재사용) ─────
	fmt.Println("▶ [7] Product 컬렉션 (제네릭 재사용 확인)")
	fmt.Println("─────────────────────────────────────────────────")

	products := NewCollection[Product]()
	productData := []Product{
		{ID: 1, Name: "MacBook Pro", Category: "Laptop", Price: 2499.99, Stock: 50},
		{ID: 2, Name: "iPhone 16", Category: "Phone", Price: 999.99, Stock: 200},
		{ID: 3, Name: "iPad Air", Category: "Tablet", Price: 749.99, Stock: 150},
		{ID: 4, Name: "AirPods Pro", Category: "Audio", Price: 249.99, Stock: 500},
		{ID: 5, Name: "Apple Watch", Category: "Wearable", Price: 399.99, Stock: 100},
	}
	for _, p := range productData {
		products.Insert(p)
	}

	// 가격 500 이상 제품, 가격 내림차순
	expensive := products.Query().
		Where(func(p Product) bool { return p.Price >= 500 }).
		OrderBy(func(a, b Product) int { return cmp.Compare(b.Price, a.Price) }).
		Execute()

	fmt.Println("  💎 고가 제품 ($500 이상):")
	for _, p := range expensive {
		fmt.Printf("     %-15s | $%8.2f | 재고: %3d\n",
			p.Name, p.Price, p.Stock)
	}

	allProducts := products.FindAll()
	totalInventoryValue := Sum(allProducts, func(p Product) float64 {
		return p.Price * float64(p.Stock)
	})
	fmt.Printf("\n  📦 총 재고 가치: $%.2f\n", totalInventoryValue)

	fmt.Println()

	// ── 8. Update / Delete 시연 ───────────────────────
	fmt.Println("▶ [8] Update / Delete 시연")
	fmt.Println("─────────────────────────────────────────────────")

	// Alice 급여 인상
	alice, _ := employees.FindByID(1)
	oldSalary := alice.Salary
	alice.Salary = 105000
	employees.Update(alice)

	updated, _ := employees.FindByID(1)
	fmt.Printf("  ✏️  Alice 급여 업데이트: $%.0f → $%.0f\n", oldSalary, updated.Salary)

	time.Sleep(50 * time.Millisecond)

	// Frank 삭제
	employees.Delete(6)
	fmt.Printf("  🗑️  Frank 삭제 후 직원 수: %d명\n", employees.Count())

	time.Sleep(50 * time.Millisecond)

	fmt.Println()
	fmt.Println("╔══════════════════════════════════════════════════╗")
	fmt.Println("║   ✅ GenDB 데모 완료                              ║")
	fmt.Println("╚══════════════════════════════════════════════════╝")
}

// ──────────────────────────────────────────────
// 유틸리티
// ──────────────────────────────────────────────

func mustParseDate(s string) time.Time {
	t, err := time.Parse("2006-01-02", s)
	if err != nil {
		panic(err)
	}
	return t
}
```

Go 제네릭스는 "타입을 추상화하는 도구"가 아니라 "타입 안전성을 희생하지 않고 코드를 재사용하는 도구"입니다. 인터페이스가 행동(메서드)을 추상화한다면, 제네릭스는 타입 자체를 파라미터화합니다. 두 도구는 경쟁 관계가 아니라 상호보완적인 관계이며, 상황에 맞게 올바른 도구를 선택하는 안목이 Go 개발자로서의 실력을 결정짓습니다.
